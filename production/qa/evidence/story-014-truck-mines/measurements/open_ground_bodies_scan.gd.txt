extends SceneTree
## Scratch scan (never in the repo): every collider on the open-ground layers (97) of Map 01, its
## thinnest horizontal extent (a box's own width, a cylinder's diameter, a cliff's bounding box) and the
## height of its bottom, to answer the engine review's open points on thin bodies and overhead geometry.
## Run: godot --headless --path . -s <this file>

const MASK: int = 97

func _initialize() -> void:
	var map: Node3D = (load("res://src/gameplay/maps/map_01/map_01.tscn") as PackedScene).instantiate() as Node3D
	root.add_child(map)
	await process_frame
	var rows: Array = []
	_walk(map, rows)
	rows.sort_custom(func(a: Array, b: Array) -> bool: return float(a[1]) < float(b[1]))
	for row: Array in rows:
		print("BODY thin=%.3f bottom=%.3f top=%.3f layer=%d kind=%s centre=%s path=%s" % [row[1], row[2], row[3], row[4], row[5], row[6], row[0]])
	var raised: int = rows.filter(func(row: Array) -> bool: return float(row[2]) > 0.05).size()
	print("SUMMARY colliders=%d raised_above_the_floor=%d thinnest=%.3f (%s)" % [rows.size(), raised, float(rows[0][1]), rows[0][0]])
	quit()


func _walk(node: Node, rows: Array) -> void:
	if node is CollisionObject3D and ((node as CollisionObject3D).collision_layer & MASK) != 0:
		for child: Node in node.get_children():
			if child is CollisionShape3D and not (child as CollisionShape3D).disabled and (child as CollisionShape3D).shape != null:
				_shape_row(child as CollisionShape3D, (node as CollisionObject3D).collision_layer, rows)
	if node is CSGShape3D and (node as CSGShape3D).use_collision and ((node as CSGShape3D).collision_layer & MASK) != 0 and (node as CSGShape3D).is_root_shape():
		var csg: CSGShape3D = node as CSGShape3D
		var box: AABB = csg.global_transform * csg.get_aabb()
		rows.append([str(csg.get_path()), minf(box.size.x, box.size.z), box.position.y, box.end.y, csg.collision_layer, "csg", box.get_center()])
	for child: Node in node.get_children():
		_walk(child, rows)


func _shape_row(node: CollisionShape3D, layer: int, rows: Array) -> void:
	var shape: Shape3D = node.shape
	var xf: Transform3D = node.global_transform
	var thin: float = INF
	var local: AABB
	if shape is BoxShape3D:
		var size: Vector3 = (shape as BoxShape3D).size * xf.basis.get_scale()
		thin = minf(size.x, size.z)
		local = AABB(-(shape as BoxShape3D).size / 2.0, (shape as BoxShape3D).size)
	elif shape is CylinderShape3D:
		var c: CylinderShape3D = shape as CylinderShape3D
		thin = c.radius * 2.0 * xf.basis.get_scale().x
		local = AABB(Vector3(-c.radius, -c.height / 2.0, -c.radius), Vector3(c.radius * 2.0, c.height, c.radius * 2.0))
	else:
		local = shape.get_debug_mesh().get_aabb()
		var world_box: AABB = xf * local
		thin = minf(world_box.size.x, world_box.size.z)
	var box: AABB = xf * local
	rows.append([str(node.get_path()), thin, box.position.y, box.end.y, layer, shape.get_class(), box.get_center()])
