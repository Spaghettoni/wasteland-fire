#!/bin/sh
# The final chain of Story 021: import, parse every .gd the story added or changed, regenerate the exact rewrites from the source
# lines, regress the 68 older outputs, run crash twice (byte compare) and the showcase headless, gdUnit, and the main scene.
S=/private/tmp/claude-501/-Users-tomas-kunzo-projects-ronin-ya-wasteland-fire/c295cfaf-3b24-4d21-bd9f-50b14eefa68f/scratchpad
cd /Users/tomas.kunzo/projects/ronin-ya/wasteland-fire || exit 9
godot --headless --path . --import > $S/w21/import.txt 2>&1; echo "import rc=$?"; grep -c -E "ERROR|WARNING" $S/w21/import.txt
{ for f in src/gameplay/units/unit.gd src/gameplay/units/unit_stats.gd tools/evidence/split_screen_harness.gd; do echo $f; done; git ls-files --others --exclude-standard -- 'tools/evidence/split_screen/crash*.gd' | grep '\.gd$'; } | sort -u > $S/w21/gd-list.txt
wc -l < $S/w21/gd-list.txt
sed 's#^#res://#' $S/w21/gd-list.txt | xargs godot --headless --path . -s res://.claude/scripts/godot-parse-check.gd -- > $S/w21/parse.txt 2>&1; echo "parse rc=$?"; grep -c "^ok:" $S/w21/parse.txt; grep "PARSE FAIL" $S/w21/parse.txt
python3 -I $S/rg21/mk_normalizers.py /Users/tomas.kunzo/projects/ronin-ya/wasteland-fire $S/rg21/f-base $S/rg21/before21 $S/rg21/normalizers21.json | tail -4
sh $S/rg21/regress.sh final $S/rg21/normalizers21.json 2>&1 | tail -14
for run in a b; do godot --headless --fixed-fps 60 --path . res://tools/evidence/split_screen_harness.tscn -- --scenario=crash > $S/gsw21/final-$run.txt 2>&1; echo "crash $run rc=$?"; done
grep -c "^CHECK" $S/gsw21/final-a.txt; grep -E "^RESULT|FAIL" $S/gsw21/final-a.txt | cut -c1-300
cmp $S/gsw21/final-a.txt $S/gsw21/final-b.txt && echo "BYTE-IDENTICAL"
godot --headless --fixed-fps 60 --path . res://tools/evidence/split_screen_harness.tscn -- --scenario=crash_showcase > $S/gsw21/showcase-headless.txt 2>&1; echo "showcase rc=$?"; grep -E "^RESULT|FAIL" $S/gsw21/showcase-headless.txt | cut -c1-200
godot --headless -s -d --remote-debug tcp://127.0.0.1:0 res://addons/gdUnit4/bin/GdUnitCmdTool.gd -a res://tests --ignoreHeadlessMode > $S/w21/gdunit.txt 2>&1; echo "gdunit rc=$?"; grep -a "Overall Summary" $S/w21/gdunit.txt | cut -c1-200
godot --path . --windowed --resolution 1280x720 --quit-after 600 > $S/w21/main-scene-600-frames.txt 2>&1; echo "main scene rc=$?"; grep -c -E "ERROR|WARNING|SCRIPT" $S/w21/main-scene-600-frames.txt
echo FINAL_DONE
