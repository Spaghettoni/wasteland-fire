def prepare(i):
    d = os.path.join(HERE, "copy%d" % i)
    if os.path.exists(d): subprocess.run(["find", d, "-delete"], check=True)
    subprocess.run(["rsync", "-a", "--exclude", ".git", REPO + "/", d + "/"], check=True)
    return d
def run(copy, scn, kind="check"):
    p = subprocess.run(["godot","--headless","--fixed-fps","60","--path",copy,"res://tools/evidence/split_screen_harness.tscn","--","--scenario="+scn],capture_output=True,text=True,timeout=600)
    return p.returncode, p.stdout + p.stderr
def one(copy, mutant):
    name, rel, old, new, scn, kind = mutant
    path = os.path.join(copy, rel)
    orig = open(path).read()          # this worker is the only one that touches this copy
    if orig.count(old) != 1:
        return name, None, "NOT APPLIED (text found %d times)" % orig.count(old)
    open(path, "w").write(orig.replace(old, new))
    try:
        rc, out = run(copy, scn, kind)
    finally:
        open(path, "w").write(orig)
    if kind == "check":
        fails = re.findall(r"^CHECK \S+ (\S+) FAIL", out, re.M)
        dead = rc != 0 or bool(fails)
        detail = "%s rc=%d fails=%s" % (scn, rc, ",".join(fails) or "-")
    else:
        base = open(os.path.join(BASE, "split-%s.txt" % scn)).read()
        keep = lambda t: [l for l in t.splitlines() if l.startswith(("SPLIT ","CHECK ","RESULT "))]
        dead = keep(out) != keep(base) or rc != 0
        detail = "%s rc=%d output %s the baseline" % (scn, rc, "differs from" if dead else "equals")
    return name, dead, detail
def worker(i, jobs):
    copy = COPIES[i]
    return [one(copy, m) for m in jobs[i::WORKERS]]
if __name__ == "__main__":
    only = set(sys.argv[1:])
    COPIES = [prepare(i) for i in range(WORKERS)]
    jobs = [m for m in M if not only or m[0] in only]
    killed = survived = 0
    with ThreadPoolExecutor(WORKERS) as ex:
        results = list(ex.map(lambda i: worker(i, jobs), range(WORKERS)))
    byname = {r[0]: r for part in results for r in part}
    for m in jobs:
        name, dead, detail = byname[m[0]]
        if dead is None:
            print("MUTANT %-42s %s" % (name, detail)); continue
        killed += 1 if dead else 0
        survived += 0 if dead else 1
        print("MUTANT %-42s %s | %s" % (name, "KILLED" if dead else "SURVIVED", detail))
    print("KILLED=%d SURVIVED=%d" % (killed, survived))
    # a clean copy must still equal the repo: the guard against the bug of the first version of this script
    for i in range(WORKERS):
        r = subprocess.run(["diff", "-rq", "--exclude=.godot", REPO + "/src", COPIES[i] + "/src"], capture_output=True, text=True)
        r2 = subprocess.run(["diff", "-rq", REPO + "/tools", COPIES[i] + "/tools"], capture_output=True, text=True)
        print("COPY%d CLEAN=%s" % (i, not (r.stdout.strip() or r2.stdout.strip())))
