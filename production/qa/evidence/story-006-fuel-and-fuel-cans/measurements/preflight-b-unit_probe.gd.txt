extends "res://src/gameplay/units/unit.gd"
## Probe-only subclass of the real Unit: emulates Story 006's DESIGN rules (crash from the Unit's own
## tick; effective throttle 0 and a fixed yaw rate while "empty") without touching the repo.

var crash_at_frame: int = -1
var empty_flag: bool = false
var empty_turn_rate: float = 0.0
var arm_empty_at_top: bool = false
var steer_on_arm: float = 0.0
var flag_frame: int = -1
var samples: Array = []
var tick_frames: Array = []
var destroy_calls: int = 0


func _physics_process(delta: float) -> void:
	var frame: int = Engine.get_physics_frames()
	tick_frames.append(frame)
	if arm_empty_at_top and is_equal_approx(_speed, stats.max_speed):
		empty_flag = true
		arm_empty_at_top = false
		flag_frame = frame
		if steer_on_arm != 0.0:
			_steer = steer_on_arm
	if crash_at_frame >= 0 and frame >= crash_at_frame and _is_alive:
		crash_at_frame = -1
		destroy_calls += 1
		destroy()
		return
	super(delta)
	var heading: Vector3 = -global_transform.basis.z
	samples.append([frame, _speed, global_position.x, global_position.y, global_position.z, atan2(-heading.x, -heading.z), is_on_floor(), get_real_velocity().length()])


func _effective_throttle() -> float:
	return 0.0 if empty_flag else _throttle


func _next_speed(delta: float) -> float:
	var thr: float = _effective_throttle()
	var speed: float = _speed
	if _blocked and signf(thr) * signf(speed) < 0.0:
		speed = 0.0
	if is_zero_approx(thr):
		return move_toward(speed, 0.0, stats.coast_deceleration * delta)
	var top_speed: float = stats.max_speed if thr > 0.0 else stats.reverse_max_speed
	var opposing: bool = signf(thr) * signf(speed) < 0.0
	var rate: float = stats.braking if opposing else stats.acceleration
	return move_toward(speed, top_speed * thr, rate * delta)


func _yaw_rate() -> float:
	if empty_flag:
		return _steer * empty_turn_rate
	var speed_fraction: float = minf(absf(_speed) / stats.max_speed, 1.0)
	return _steer * stats.turn_rate * speed_fraction * signf(_speed)
