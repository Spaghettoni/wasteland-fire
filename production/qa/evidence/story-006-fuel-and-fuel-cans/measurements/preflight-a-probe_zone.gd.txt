extends SceneTree
## Preflight P1 for Story 006: Fuel Can zone polling. Prints PROBE lines only.

const UNIT_SCENE: PackedScene = preload("res://src/gameplay/units/motorbike.tscn")
const SPLIT_PATH: String = "res://src/gameplay/split_screen/split_screen.tscn"
const STATS_PATHS: Dictionary = {
	"motorbike": "res://src/gameplay/units/data/motorbike_stats.tres",
	"buggy": "res://src/gameplay/units/data/buggy_stats.tres",
	"truck": "res://src/gameplay/units/data/truck_stats.tres",
	"gyrocopter": "res://src/gameplay/units/data/gyrocopter_stats.tres",
}
const ZONE_RADIUS: float = 1.6
const ZONE_LIFT: float = 0.6
const ZONE_MASK: int = 18
const PARK: Vector3 = Vector3(30.0, 0.05, 30.0)
const WATCHDOG_TICKS: int = 40000

var _only: PackedStringArray = PackedStringArray()
var _ticks: int = 0


class Poller extends Node:
	var sampler: Callable
	var zone: Area3D
	var unit: Unit
	var tag: String = ""
	var sink: Array = []

	func _physics_process(_delta: float) -> void:
		sink.append(sampler.call(zone, unit, tag))


func _physics_process(_delta: float) -> bool:
	_ticks += 1
	if _ticks > WATCHDOG_TICKS:
		print("PROBE watchdog tripped ticks=%d" % _ticks)
		return true
	return false


func _initialize() -> void:
	for a: String in OS.get_cmdline_user_args():
		if a.begins_with("--only="):
			_only = a.substr(7).split(",")
	var info: Dictionary = Engine.get_version_info()
	_p("env", "godot=%s physics_ticks=%d physics_engine=%s args=%s" % [info["string"], Engine.physics_ticks_per_second, str(ProjectSettings.get_setting("physics/3d/physics_engine")), " ".join(OS.get_cmdline_args())])
	await _run_all()
	_p("done", "ticks=%d" % _ticks)
	quit(0)


func _p(id: String, text: String) -> void:
	print("PROBE %s %s" % [id, text])


func _want(id: String) -> bool:
	return _only.is_empty() or _only.has(id)


func _run_all() -> void:
	if _want("a"):
		for t: String in ["truck", "buggy", "motorbike"]:
			await _phase_pass(t, "a")
	if _want("b"):
		await _phase_pass("gyrocopter", "b")
	if _want("c"):
		for t: String in ["truck", "gyrocopter"]:
			await _phase_teleport(t)
	if _want("d"):
		await _phase_layer_zero()
	if _want("e"):
		await _phase_order()
	if _want("f"):
		await _phase_persist()
	if _want("g"):
		await _phase_split()
	if _want("e2"):
		await _phase_order_many()
	if _want("f2"):
		for t2: String in ["truck", "gyrocopter"]:
			await _one_rest(t2, false)
			await _one_rest(t2, true)
	if _want("h"):
		await _phase_pause_ghost()


# ---------------------------------------------------------------- builders

func _new_world() -> Node3D:
	var w: Node3D = Node3D.new()
	w.name = "ProbeWorld"
	root.add_child(w)
	var floor_body: StaticBody3D = StaticBody3D.new()
	floor_body.name = "Floor"
	floor_body.collision_layer = 1
	floor_body.collision_mask = 0
	var cs: CollisionShape3D = CollisionShape3D.new()
	var box: BoxShape3D = BoxShape3D.new()
	box.size = Vector3(80.0, 1.0, 80.0)
	cs.shape = box
	cs.position = Vector3(0.0, -0.5, 0.0)
	floor_body.add_child(cs)
	w.add_child(floor_body)
	return w


func _new_can(parent: Node, at: Vector3) -> Node3D:
	var can: Node3D = Node3D.new()
	can.name = "Can"
	can.position = at
	var body: Node3D = Node3D.new()
	body.name = "Body"
	var mesh: MeshInstance3D = MeshInstance3D.new()
	mesh.mesh = BoxMesh.new()
	body.add_child(mesh)
	can.add_child(body)
	var zone: Area3D = Area3D.new()
	zone.name = "PickupZone"
	zone.collision_layer = 0
	zone.collision_mask = ZONE_MASK
	var cs: CollisionShape3D = CollisionShape3D.new()
	var sphere: SphereShape3D = SphereShape3D.new()
	sphere.radius = ZONE_RADIUS
	cs.shape = sphere
	cs.position = Vector3(0.0, ZONE_LIFT, 0.0)
	zone.add_child(cs)
	can.add_child(zone)
	parent.add_child(can)
	return can


func _new_unit(parent: Node, uname: String, at: Vector3) -> Unit:
	var u: Unit = UNIT_SCENE.instantiate() as Unit
	u.name = uname
	u.position = at
	parent.add_child(u)
	return u


func _new_poller(parent: Node, zone: Area3D, unit: Unit, tag: String, prio: int) -> Poller:
	var p: Poller = Poller.new()
	p.name = "Poll_" + tag
	p.sampler = _sample
	p.zone = zone
	p.unit = unit
	p.tag = tag
	p.process_physics_priority = prio
	parent.add_child(p)
	return p


func _free_world(w: Node) -> void:
	w.queue_free()


func _stats(type_id: String) -> UnitStats:
	return load(STATS_PATHS[type_id]) as UnitStats


# ---------------------------------------------------------------- sampling

func _shape_node(unit: Unit) -> CollisionShape3D:
	for child: Node in unit.get_children():
		if child is CollisionShape3D:
			return child as CollisionShape3D
	return null


func _geo_dist(zone: Area3D, unit: Unit) -> float:
	var us: CollisionShape3D = _shape_node(unit)
	var zs: CollisionShape3D = zone.get_child(0) as CollisionShape3D
	var box: BoxShape3D = us.shape as BoxShape3D
	var local: Vector3 = us.global_transform.affine_inverse() * zs.global_position
	var half: Vector3 = box.size * 0.5
	var q: Vector3 = Vector3(maxf(absf(local.x) - half.x, 0.0), maxf(absf(local.y) - half.y, 0.0), maxf(absf(local.z) - half.z, 0.0))
	return q.length()


func _sample(zone: Area3D, unit: Unit, tag: String) -> Dictionary:
	var bodies: Array[Node3D] = zone.get_overlapping_bodies()
	var names: PackedStringArray = PackedStringArray()
	for b: Node3D in bodies:
		names.append(String(b.name))
	var dist: float = _geo_dist(zone, unit)
	var pos: Vector3 = unit.global_position
	return {
		"t": Engine.get_physics_frames(), "tag": tag, "gob": bodies.has(unit),
		"ob": zone.overlaps_body(unit), "n": bodies.size(), "dist": dist,
		"geo": dist < ZONE_RADIUS, "x": pos.x, "y": pos.y, "z": pos.z,
		"spd": unit.current_speed, "names": ",".join(names),
	}


func _stream_stats(samples: Array, t_from: int) -> Dictionary:
	var geo_first: int = -1
	var geo_last: int = -1
	var gob_first: int = -1
	var gob_last: int = -1
	var gob_n: int = 0
	var ob_first: int = -1
	var ob_last: int = -1
	var ob_n: int = 0
	var ticks: int = 0
	var mismatch: int = 0
	var n_max: int = 0
	var spd: float = 0.0
	var dist_first: float = 0.0
	var dist_last: float = 0.0
	for s: Dictionary in samples:
		var t: int = s["t"]
		if t < t_from:
			continue
		ticks += 1
		if s["geo"]:
			if geo_first < 0:
				geo_first = t
				spd = s["spd"]
			geo_last = t
		if s["gob"]:
			if gob_first < 0:
				gob_first = t
				dist_first = s["dist"]
			gob_last = t
			dist_last = s["dist"]
			gob_n += 1
		if s["ob"]:
			if ob_first < 0:
				ob_first = t
			ob_last = t
			ob_n += 1
		if bool(s["gob"]) != bool(s["ob"]):
			mismatch += 1
		n_max = maxi(n_max, int(s["n"]))
	return {"geo_first": geo_first, "geo_last": geo_last, "gob_first": gob_first, "gob_last": gob_last,
		"gob_n": gob_n, "ob_first": ob_first, "ob_last": ob_last, "ob_n": ob_n, "ticks": ticks,
		"mismatch": mismatch, "n_max": n_max, "spd": spd, "dist_first": dist_first, "dist_last": dist_last}


func _fmt(tag: String, st: Dictionary, ref: Dictionary) -> String:
	return "%s geo=[%d..%d] gob=[%d..%d]x%d ob=[%d..%d]x%d | lag_in vs own geo=%d, vs pf geo=%d | lag_out vs own geo=%d, vs pf geo=%d | ob_lag_in=%d | pen_in=%.3f pen_out=%.3f spd_at_geo=%.2f mismatch=%d n_max=%d" % [
		tag, st["geo_first"], st["geo_last"], st["gob_first"], st["gob_last"], st["gob_n"],
		st["ob_first"], st["ob_last"], st["ob_n"],
		st["gob_first"] - st["geo_first"], st["gob_first"] - ref["geo_first"],
		st["gob_last"] - st["geo_last"], st["gob_last"] - ref["geo_last"],
		st["ob_first"] - st["geo_first"],
		ZONE_RADIUS - st["dist_first"], ZONE_RADIUS - st["dist_last"], st["spd"], st["mismatch"], st["n_max"]]


func _first_tick(samples: Array, key: String, want: bool, t_from: int, t_to: int) -> int:
	for s: Dictionary in samples:
		var t: int = s["t"]
		if t >= t_from and t < t_to and bool(s[key]) == want:
			return t
	return -1


func _count_true(samples: Array, key: String, t_from: int, t_to: int) -> int:
	var n: int = 0
	for s: Dictionary in samples:
		var t: int = s["t"]
		if t >= t_from and t < t_to and bool(s[key]):
			n += 1
	return n


# ---------------------------------------------------------------- (a) (b) pass through

func _phase_pass(type_id: String, id: String) -> void:
	var w: Node3D = _new_world()
	var can: Node3D = _new_can(w, Vector3.ZERO)
	var zone: Area3D = can.get_node("PickupZone") as Area3D
	var unit: Unit = _new_unit(w, "Unit", PARK)
	var pb: Poller = _new_poller(w, zone, unit, "pb", -100)
	var pa: Poller = _new_poller(w, zone, unit, "pa", 100)
	var stats_res: UnitStats = _stats(type_id)
	await physics_frame
	unit.leave_play()
	await physics_frame
	unit.spawn(Transform3D(Basis.IDENTITY, Vector3(0.0, 0.0, 26.0)), stats_res)
	unit.set_drive_input(1.0, 0.0)
	var t_spawn: int = Engine.get_physics_frames()
	var pf: Array = []
	for i: int in 500:
		await physics_frame
		pf.append(_sample(zone, unit, "pf"))
		if unit.global_position.z < -14.0:
			break
	await physics_frame
	var label: String = "%s.%s" % [id, type_id]
	var st_pf: Dictionary = _stream_stats(pf, t_spawn)
	var st_pb: Dictionary = _stream_stats(pb.sink, t_spawn)
	var st_pa: Dictionary = _stream_stats(pa.sink, t_spawn)
	var shape: Shape3D = _shape_node(unit).shape
	_p(label, "cfg max_speed=%.2f box=%s per_tick_m=%.3f layer=%d mask=%d ticks=%d" % [stats_res.max_speed, str((shape as BoxShape3D).size), stats_res.max_speed / 60.0, unit.collision_layer, unit.collision_mask, st_pf["ticks"]])
	_p(label, _fmt("pf(physics_frame continuation)", st_pf, st_pf))
	_p(label, _fmt("pb(node before Unit)", st_pb, st_pf))
	_p(label, _fmt("pa(node after Unit)", st_pa, st_pf))
	_free_world(w)
	await physics_frame
	await physics_frame


# ---------------------------------------------------------------- (c) teleport in and out

func _phase_teleport(type_id: String) -> void:
	var w: Node3D = _new_world()
	var can: Node3D = _new_can(w, Vector3.ZERO)
	var zone: Area3D = can.get_node("PickupZone") as Area3D
	var unit: Unit = _new_unit(w, "Unit", PARK)
	var pb: Poller = _new_poller(w, zone, unit, "pb", -100)
	var pa: Poller = _new_poller(w, zone, unit, "pa", 100)
	var stats_res: UnitStats = _stats(type_id)
	var far_xf: Transform3D = Transform3D(Basis.IDENTITY, Vector3(0.0, 0.0, 20.0))
	var in_xf: Transform3D = Transform3D(Basis.IDENTITY, Vector3.ZERO)
	await physics_frame
	unit.leave_play()
	await physics_frame
	unit.spawn(far_xf, stats_res)
	var t_start: int = Engine.get_physics_frames()
	var pf: Array = []
	var t_in: int = -1
	var t_out: int = -1
	for i: int in 90:
		await physics_frame
		pf.append(_sample(zone, unit, "pf"))
		if i == 20:
			t_in = Engine.get_physics_frames()
			unit.spawn(in_xf, stats_res)
		elif i == 55:
			t_out = Engine.get_physics_frames()
			unit.spawn(far_xf, stats_res)
	await physics_frame
	var label: String = "c.%s" % type_id
	var streams: Dictionary = {"pf": pf, "pb": pb.sink, "pa": pa.sink}
	for key: String in ["pf", "pb", "pa"]:
		var smp: Array = streams[key]
		var in_g: int = _first_tick(smp, "gob", true, t_in, t_out)
		var in_o: int = _first_tick(smp, "ob", true, t_in, t_out)
		var out_g: int = _first_tick(smp, "gob", false, t_out + 1, t_out + 40)
		var out_o: int = _first_tick(smp, "ob", false, t_out + 1, t_out + 40)
		var pre_listed: int = _count_true(smp, "gob", t_start, t_in)
		var geo_in: int = _first_tick(smp, "geo", true, t_in, t_out)
		_p(label, "%s: spawn-into-zone called at t=%d: first geo(at that stream's sample)=t+%d, first reported gob=t+%d ob=t+%d | spawn-out called at t=%d: first unreported gob=t+%d ob=t+%d | listed_before_teleport_ticks=%d" % [key, t_in, geo_in - t_in, in_g - t_in, in_o - t_in, t_out, out_g - t_out, out_o - t_out, pre_listed])
	_free_world(w)
	await physics_frame
	await physics_frame


# ---------------------------------------------------------------- (d) layer and mask zero

func _phase_layer_zero() -> void:
	for mode: String in ["destroy", "bench"]:
		var w: Node3D = _new_world()
		var can: Node3D = _new_can(w, Vector3.ZERO)
		var zone: Area3D = can.get_node("PickupZone") as Area3D
		var unit: Unit = _new_unit(w, "Unit", PARK)
		var pb: Poller = _new_poller(w, zone, unit, "pb", -100)
		var pa: Poller = _new_poller(w, zone, unit, "pa", 100)
		var stats_res: UnitStats = _stats("truck")
		var far_xf: Transform3D = Transform3D(Basis.IDENTITY, Vector3(0.0, 0.0, 20.0))
		var in_xf: Transform3D = Transform3D(Basis.IDENTITY, Vector3.ZERO)
		await physics_frame
		unit.leave_play()
		await physics_frame
		unit.spawn(far_xf, stats_res)
		var pf: Array = []
		var t_x: int = -1
		var t_walk: int = -1
		var t_rev: int = -1
		var layers_at_walk: String = ""
		var t_start: int = Engine.get_physics_frames()
		for i: int in 380:
			await physics_frame
			pf.append(_sample(zone, unit, "pf"))
			if i == 5:
				unit.spawn(in_xf, stats_res)
			elif i == 30:
				t_x = Engine.get_physics_frames()
				if mode == "destroy":
					unit.destroy()
				else:
					unit.leave_play()
			elif i == 180:
				unit.global_position = PARK
			elif i == 200:
				t_walk = Engine.get_physics_frames()
				unit.global_position = Vector3.ZERO
				layers_at_walk = "layer=%d mask=%d alive=%s visible=%s" % [unit.collision_layer, unit.collision_mask, str(unit.is_alive), str(unit.visible)]
			elif i == 350:
				t_rev = Engine.get_physics_frames()
				unit.spawn(in_xf, stats_res)
		await physics_frame
		var streams: Dictionary = {"pf": pf, "pb": pb.sink, "pa": pa.sink}
		for key: String in ["pf", "pb", "pa"]:
			var smp: Array = streams[key]
			var listed_before: int = _count_true(smp, "gob", t_x - 10, t_x)
			var exit_g: int = _first_tick(smp, "gob", false, t_x + 1, t_walk)
			var exit_o: int = _first_tick(smp, "ob", false, t_x + 1, t_walk)
			var after_exit: int = _count_true(smp, "gob", exit_g, t_walk - 15)
			var walk_in: int = _count_true(smp, "gob", t_walk, t_rev)
			var walk_in_ob: int = _count_true(smp, "ob", t_walk, t_rev)
			var rev_g: int = _first_tick(smp, "gob", true, t_rev, t_rev + 30)
			_p("d.%s" % mode, "%s: listed_in_10_ticks_before_call=%d | exit_lag gob=%d ob=%d ticks after the call | listed_ticks_after_exit(while layer=0, standing inside)=%d | walk-in with layer 0 for %d ticks listed gob=%d ob=%d | revive spawn(): first listed t+%d | at walk-in: %s" % [key, listed_before, exit_g - t_x, exit_o - t_x, after_exit, t_rev - t_walk, walk_in, walk_in_ob, rev_g - t_rev, layers_at_walk])
		_free_world(w)
		await physics_frame
		await physics_frame


# ---------------------------------------------------------------- (e) order of the list

func _one_order(pair: String, add_order: String, spawn_order: String, swap: bool = false) -> Dictionary:
	var w: Node3D = _new_world()
	var can: Node3D = _new_can(w, Vector3.ZERO)
	var zone: Area3D = can.get_node("PickupZone") as Area3D
	var second_type: String = "buggy" if pair == "truck_buggy" else "gyrocopter"
	var units: Dictionary = {}
	for key: String in add_order:
		var at: Vector3 = PARK if key == "A" else Vector3(-30.0, 0.05, 30.0)
		units[key] = _new_unit(w, "U" + key, at)
	await physics_frame
	for key: String in ["A", "B"]:
		(units[key] as Unit).leave_play()
	await physics_frame
	var stats_a: UnitStats = _stats("truck")
	var stats_b: UnitStats = _stats(second_type)
	var pos_a: Vector3 = Vector3(0.0, 0.0, -2.6) if pair == "truck_buggy" else Vector3.ZERO
	var pos_b: Vector3 = Vector3(0.0, 0.0, 2.0) if pair == "truck_buggy" else Vector3.ZERO
	if swap:
		var tmp: Vector3 = pos_a
		pos_a = pos_b
		pos_b = tmp
	var t_spawn: int = Engine.get_physics_frames()
	for key: String in spawn_order:
		if key == "A":
			(units["A"] as Unit).spawn(Transform3D(Basis.IDENTITY, pos_a), stats_a)
		else:
			(units["B"] as Unit).spawn(Transform3D(Basis.IDENTITY, pos_b), stats_b)
	var first_a: int = -1
	var first_b: int = -1
	var both_t: int = -1
	var order: String = ""
	var inst: String = ""
	for i: int in 12:
		await physics_frame
		var bodies: Array[Node3D] = zone.get_overlapping_bodies()
		var names: PackedStringArray = PackedStringArray()
		for b: Node3D in bodies:
			names.append(String(b.name))
		var t: int = Engine.get_physics_frames()
		if first_a < 0 and names.has("UA"):
			first_a = t
		if first_b < 0 and names.has("UB"):
			first_b = t
		if both_t < 0 and names.has("UA") and names.has("UB"):
			both_t = t
			order = ",".join(names)
			inst = "A_id<B_id=%s" % str((units["A"] as Unit).get_instance_id() < (units["B"] as Unit).get_instance_id())
	_free_world(w)
	await physics_frame
	await physics_frame
	return {"first_a": first_a, "first_b": first_b, "both_t": both_t, "order": order, "t_spawn": t_spawn, "inst": inst}


func _phase_order() -> void:
	var table: Dictionary = {}
	for pair: String in ["truck_buggy", "truck_gyro"]:
		for add_order: String in ["AB", "BA"]:
			for spawn_order: String in ["AB", "BA"]:
				for run_i: int in 3:
					var r: Dictionary = await _one_order(pair, add_order, spawn_order)
					var key: String = "%s add_to_tree=%s spawn_called=%s" % [pair, add_order, spawn_order]
					if not table.has(key):
						table[key] = []
					(table[key] as Array).append(r)
	for key: String in table.keys():
		var rs: Array = table[key]
		var parts: PackedStringArray = PackedStringArray()
		for r: Dictionary in rs:
			parts.append("[%s A@+%d B@+%d %s]" % [r["order"], int(r["first_a"]) - int(r["t_spawn"]), int(r["first_b"]) - int(r["t_spawn"]), r["inst"]])
		_p("e", "%s -> %s" % [key, " ".join(parts)])


# ---------------------------------------------------------------- (f) persistence

func _one_persist(type_id: String, blind: bool) -> void:
	var w: Node3D = _new_world()
	var can: Node3D = _new_can(w, Vector3.ZERO)
	var zone: Area3D = can.get_node("PickupZone") as Area3D
	var body: Node3D = can.get_node("Body") as Node3D
	var unit: Unit = _new_unit(w, "Unit", PARK)
	var stats_res: UnitStats = _stats(type_id)
	await physics_frame
	unit.leave_play()
	await physics_frame
	unit.spawn(Transform3D(Basis.IDENTITY, Vector3.ZERO), stats_res)
	var t_spawn: int = Engine.get_physics_frames()
	var first_listed: int = -1
	for i: int in 30:
		await physics_frame
		if zone.overlaps_body(unit):
			first_listed = Engine.get_physics_frames()
			break
	var start_pos: Vector3 = unit.global_position
	var total: int = 1300
	var hide_at: int = 50 if blind else 100
	var listed: int = 0
	var unlisted: int = 0
	var first_gap: int = -1
	var ob_gap: int = 0
	var max_n: int = 0
	var sampled: int = 0
	for i: int in total + 30:
		await physics_frame
		if i == hide_at:
			body.visible = false
		if blind and i < total:
			continue
		var bodies: Array[Node3D] = zone.get_overlapping_bodies()
		sampled += 1
		max_n = maxi(max_n, bodies.size())
		if bodies.has(unit):
			listed += 1
		else:
			unlisted += 1
			if first_gap < 0:
				first_gap = i
		if not zone.overlaps_body(unit):
			ob_gap += 1
	_p("f.%s" % type_id, "mode=%s first_listed=t_spawn+%d sampled_ticks=%d listed=%d unlisted=%d first_unlisted_i=%d ob_false=%d max_list_len=%d sibling_Body_visible=%s zone.monitoring=%s moved=%.4f m blind_window_ticks=%d" % ["blind(read once after %d ticks, then 30 ticks)" % total if blind else "polled every tick for %d ticks" % total, first_listed - t_spawn, sampled, listed, unlisted, first_gap, ob_gap, max_n, str(body.visible), str(zone.monitoring), unit.global_position.distance_to(start_pos), total if blind else 0])
	_free_world(w)
	await physics_frame
	await physics_frame

func _phase_split() -> void:
	var split: SplitScreen = (load(SPLIT_PATH) as PackedScene).instantiate() as SplitScreen
	root.add_child(split)
	await physics_frame
	await physics_frame
	var world: Node3D = split.get_node("World") as Node3D
	(world.get_node("Player1DriveInput") as Node).set_physics_process(false)
	var mc: MatchController = split.match_controller
	var unit: Unit = split.player_1_unit
	var accepted: bool = mc.choose(0, 2)
	var waited: int = 0
	while not unit.is_alive and waited < 600:
		await physics_frame
		waited += 1
	_p("g.split", "choose_accepted=%s waited_ticks=%d alive=%s type=%s pos=%s basis_z=%s" % [str(accepted), waited, str(unit.is_alive), str(unit.type_id), str(unit.global_position), str(unit.global_transform.basis.z)])
	for i: int in 10:
		await physics_frame
	var facing: Vector3 = -unit.global_transform.basis.z
	facing.y = 0.0
	facing = facing.normalized()
	var zone_at: Vector3 = unit.global_position + facing * 20.0
	zone_at.y = 0.0
	var can: Node3D = _new_can(world, zone_at)
	var zone: Area3D = can.get_node("PickupZone") as Area3D
	var pb: Poller = _new_poller(world, zone, unit, "pb", -100)
	var pa: Poller = _new_poller(world, zone, unit, "pa", 100)
	await physics_frame
	var t_start: int = Engine.get_physics_frames()
	unit.set_drive_input(1.0, 0.0)
	var pf: Array = []
	for i: int in 600:
		await physics_frame
		pf.append(_sample(zone, unit, "pf"))
		if facing.dot(unit.global_position - zone_at) > 12.0:
			break
	await physics_frame
	var st_pf: Dictionary = _stream_stats(pf, t_start)
	var st_pb: Dictionary = _stream_stats(pb.sink, t_start)
	var st_pa: Dictionary = _stream_stats(pa.sink, t_start)
	_p("g.split", "zone_at=%s facing=%s ticks=%d final_pos=%s" % [str(zone_at), str(facing), st_pf["ticks"], str(unit.global_position)])
	_p("g.split", _fmt("pf(physics_frame continuation)", st_pf, st_pf))
	_p("g.split", _fmt("pb(node before Unit)", st_pb, st_pf))
	_p("g.split", _fmt("pa(node after Unit)", st_pa, st_pf))
	split.queue_free()
	await physics_frame


func _phase_order_many() -> void:
	var runs: int = 40
	for pair: String in ["truck_buggy", "truck_gyro"]:
		for add_order: String in ["AB", "BA"]:
			for spawn_order: String in ["AB", "BA"]:
				var swaps: Array = [false, true] if pair == "truck_buggy" else [false]
				for swap: bool in swaps:
					var counts: Dictionary = {}
					var split_ticks: int = 0
					for run_i: int in runs:
						var r: Dictionary = await _one_order(pair, add_order, spawn_order, swap)
						var k: String = str(r["order"])
						counts[k] = int(counts.get(k, 0)) + 1
						if int(r["first_a"]) != int(r["first_b"]):
							split_ticks += 1
					_p("e2", "%s add_to_tree=%s spawn_called=%s swap_positions=%s -> %s (A and B first listed on different ticks in %d of %d runs)" % [pair, add_order, spawn_order, str(swap), str(counts), split_ticks, runs])


func _one_rest(type_id: String, blind: bool) -> void:
	var w: Node3D = _new_world()
	var can: Node3D = _new_can(w, Vector3.ZERO)
	var zone: Area3D = can.get_node("PickupZone") as Area3D
	var body: Node3D = can.get_node("Body") as Node3D
	var unit: Unit = _new_unit(w, "Unit", PARK)
	var stats_res: UnitStats = _stats(type_id)
	await physics_frame
	unit.leave_play()
	await physics_frame
	unit.spawn(Transform3D(Basis.IDENTITY, Vector3(0.0, 0.0, 8.0)), stats_res)
	unit.set_drive_input(1.0, 0.0)
	var released: bool = false
	var rest_ticks: int = -1
	var listed_while_moving: int = 0
	for i: int in 600:
		await physics_frame
		if zone.overlaps_body(unit):
			listed_while_moving += 1
		if not released and unit.global_position.z <= 5.5:
			unit.set_drive_input(0.0, 0.0)
			released = true
		elif released and is_zero_approx(unit.current_speed):
			rest_ticks = i
			break
	var rest_pos: Vector3 = unit.global_position
	var inside: bool = _geo_dist(zone, unit) < ZONE_RADIUS
	var total: int = 1300
	var listed: int = 0
	var unlisted: int = 0
	var first_gap: int = -1
	for i: int in total + 30:
		await physics_frame
		if i == 50:
			body.visible = false
		if blind and i < total:
			continue
		if zone.get_overlapping_bodies().has(unit):
			listed += 1
		else:
			unlisted += 1
			if first_gap < 0:
				first_gap = i
	_p("f2.%s" % type_id, "drove in (throttle released at z<=5.5, coasted to rest after %d ticks at z=%.2f, box overlaps sphere=%s, listed while moving=%d ticks), then %s: sampled=%d listed=%d unlisted=%d first_unlisted_i=%d moved=%.4f m" % [rest_ticks, rest_pos.z, str(inside), listed_while_moving, "blind %d ticks then read" % total if blind else "polled every tick for %d ticks" % total, listed + unlisted, listed, unlisted, first_gap, unit.global_position.distance_to(rest_pos)])
	_free_world(w)
	await physics_frame
	await physics_frame


func _phase_pause_ghost() -> void:
	var w: Node3D = _new_world()
	var can: Node3D = _new_can(w, Vector3.ZERO)
	var zone: Area3D = can.get_node("PickupZone") as Area3D
	var unit: Unit = _new_unit(w, "Unit", PARK)
	var stats_res: UnitStats = _stats("truck")
	await physics_frame
	unit.leave_play()
	await physics_frame
	unit.spawn(Transform3D(Basis.IDENTITY, Vector3.ZERO), stats_res)
	for i: int in 20:
		await physics_frame
	var listed_before: bool = zone.get_overlapping_bodies().has(unit)
	var t_pause: int = Engine.get_physics_frames()
	paused = true
	unit.leave_play()
	var listed_during: int = 0
	for i: int in 30:
		await physics_frame
		if zone.get_overlapping_bodies().has(unit):
			listed_during += 1
	var t_unpause: int = Engine.get_physics_frames()
	paused = false
	var first_unlisted: int = -1
	for i: int in 10:
		await physics_frame
		if first_unlisted < 0 and not zone.get_overlapping_bodies().has(unit):
			first_unlisted = Engine.get_physics_frames()
	_p("h.pause_ghost", "listed before=%s | Unit benched with leave_play() in the tick the tree was paused (t=%d): still listed on %d of 30 paused ticks | unpaused in tick %d: first tick not listed = unpause tick + %d" % [str(listed_before), t_pause, listed_during, t_unpause, first_unlisted - t_unpause])
	_free_world(w)
	await physics_frame
	await physics_frame
