#!/usr/bin/env python3
"""Reads Story 026's A/B outputs (fps26/ab/fps-KIND-N.txt and gpu-KIND-N.txt): per run, the RESULT figures, the seconds that were not capped by
the display (a rate over the refresh rate: the window was covered or lost focus, so the display did not pace it), and the seconds with a frame
over the 16.6 ms budget, each with its time and phase; then the GPU summaries. Observations only, never a verdict.
usage: fps26-read.py DIR"""
import glob, os, re, sys

D = sys.argv[1]
def fields(line):
    return dict(re.findall(r"(\w+)=([^\s]+)", line))
print("%-10s %6s %6s %8s %9s %9s %6s %8s %7s %s" % ("run", "fpsmin", "steady", "ms_steady", "ms_max", "first/last", "calls", "prims", "uncapped", "long seconds (t, phase, fps, ms)"))
for path in sorted(glob.glob(os.path.join(D, "fps-*.txt"))):
    text = open(path).read()
    res = [l for l in text.splitlines() if l.startswith("RESULT")]
    if not res:
        print("%-10s no RESULT" % os.path.basename(path)[4:-4]); continue
    f = fields(res[0])
    splits = [fields(l) for l in text.splitlines() if l.startswith("SPLIT")]
    uncapped = [s for s in splits if float(s["fps"]) > float(f["refresh_hz"]) + 1.0]
    long_ = [s for s in splits if float(s["frame_ms_max"]) > 16.6]
    first_uncapped = uncapped[0]["t"] if uncapped else "-"
    print("%-10s %6s %6s %8s %9s %9s %6s %8s %4d@%-5s %s" % (os.path.basename(path)[4:-4], f["fps_min"], f["fps_min_steady"], f["frame_ms_max_steady"], f["frame_ms_max"],
          f["fps_first_window"] + "/" + f["fps_last_window"], f["draw_calls_max"], f["primitives_max"], len(uncapped), first_uncapped,
          "; ".join("%s %s st%s %s %s" % (s["t"], s["phase"], s["staging"], s["fps"], s["frame_ms_max"]) for s in long_)))
    ahead = re.search(r"never_ahead (PASS|FAIL).*?on all (\d+) tyre-frames judged: the track lagged it by (\S+) to (\S+) m", text)
    if ahead:
        print("%-10s never_ahead %s over %s tyre-frames, lag %s to %s m" % ("", ahead.group(1), ahead.group(2), ahead.group(3), ahead.group(4)))
for path in sorted(glob.glob(os.path.join(D, "gpu-*.txt"))):
    text = open(path).read()
    summ = [l for l in text.splitlines() if l.startswith("GPUTIME_SUMMARY")]
    print("%-10s %s" % (os.path.basename(path)[:-4], summ[0] if summ else "no summary"))
