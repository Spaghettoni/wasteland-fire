#!/bin/sh
# usage: fps26b.sh [ROUNDS]   Story 026's frame-rate A/B, second attempt: the first (fps26.sh, 08:46 to 08:57) was set aside, because the machine
# was in use and each run's window was covered part-way (the rate jumped from the display's 120 to an unpaced 145 fps, the one long frame of each
# run falling in the second before). Here every window is kept on top (--always-on-top), and the seconds since the user's last input are logged
# before and after each run (HIDIdleTime), so a run disturbed again can be told. ROUNDS rounds (default 3) of on (the repo), off (the repo with
# TRACKS_FPS_OFF=1: the node removed, the ground shader reading a strength of 0) and base (ab26/base: the tree before Story 026), the order turned
# each round; then the GPU time on Vulkan (gpu_time.gd, the map-art tool) of each, twice, the order turned again. One run at a time.
S=/private/tmp/claude-501/-Users-tomas-kunzo-projects-ronin-ya-wasteland-fire/903467d6-8576-4a19-ad15-5c4b7052d4e0/scratchpad
REPO=/Users/tomas.kunzo/projects/ronin-ya/wasteland-fire
BASE=$S/ab26/base
O=$S/fps26/ab
mkdir -p $O
P=$S/fps26/ab.progress
ROUNDS=${1:-3}
date +%H:%M:%S > $P
idle() { ioreg -c IOHIDSystem | awk '/HIDIdleTime/ {print int($NF/1000000000); exit}'; }
run() { # run KIND N [gpu]
  if [ "$1" = base ]; then DIR=$BASE; else DIR=$REPO; fi
  if [ "$1" = off ]; then OFF=1; else OFF=; fi
  BEFORE=$(idle)
  if [ "$3" = gpu ]; then
    (cd $DIR && env TRACKS_FPS_OFF=$OFF godot --path . --windowed --always-on-top --resolution 1280x720 --rendering-driver vulkan -s $S/ab26/gpu_time.gd -- --scenario=tyre_tracks_fps > $O/gpu-$1-$2.txt 2>&1)
    echo "gpu $1 $2 rc=$? $(date +%H:%M:%S) idle_before=${BEFORE}s idle_after=$(idle)s" >> $P
  else
    (cd $DIR && env TRACKS_FPS_OFF=$OFF godot --path . --windowed --always-on-top --resolution 1280x720 res://tools/evidence/split_screen_harness.tscn -- --scenario=tyre_tracks_fps > $O/fps-$1-$2.txt 2>&1)
    echo "fps $1 $2 rc=$? $(date +%H:%M:%S) idle_before=${BEFORE}s idle_after=$(idle)s" >> $P
  fi
}
n=1
while [ $n -le $ROUNDS ]; do
  case $((n % 3)) in
    1) run on $n; run off $n; run base $n;;
    2) run off $n; run base $n; run on $n;;
    0) run base $n; run on $n; run off $n;;
  esac
  n=$((n + 1))
done
run on 1 gpu; run off 1 gpu; run base 1 gpu
run base 2 gpu; run off 2 gpu; run on 2 gpu
echo ALLDONE >> $P
