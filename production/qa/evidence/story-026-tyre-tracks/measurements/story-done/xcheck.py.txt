#!/usr/bin/env python3
"""Cross-check: every identifier the evidence document's acceptance table cites in backticks, against the re-run's output.
A `CHECK tyre_tracks <name> PASS ...` line has the scenario first, then the check name. Prints PASS lines found, and what else the
table cites (other scenarios, mutants, files) so each can be read by hand. Reads only."""
import re, sys
MY = "/private/tmp/claude-501/-Users-tomas-kunzo-projects-ronin-ya-wasteland-fire/2573aa3b-8b37-4178-bdbd-5cb6686a7ba2/scratchpad/sd26"
doc = open("production/qa/evidence/story-026-tyre-tracks-evidence.md").read()
out = open(MY + "/out/tyre_tracks-a.txt").read()
checks = {}
for m in re.finditer(r"^CHECK (\S+) (\S+) (PASS|FAIL)\b", out, re.M):
    checks[(m.group(1), m.group(2))] = m.group(3)
names = {n for (s, n) in checks}
print("CHECK lines in the re-run:", len(checks), " all PASS:", all(v == "PASS" for v in checks.values()))
table = []
inside = False
for line in doc.split("\n"):
    if line.startswith("| AC-") or line.startswith("| AC "):
        table.append(line)
    if line.startswith("## Acceptance criteria"):
        inside = True
rows = [l for l in table if l.count("|") >= 4]
print("acceptance rows:", len(rows))
cited_pass, cited_other = [], []
for row in rows:
    ac = row.split("|")[1].strip()[:4]
    where = row.split("|")[-2]
    for tok in re.findall(r"`([^`]+)`", where):
        base = tok.split("'")[0].strip()
        if re.fullmatch(r"[a-z0-9_]+", base):
            if base in names and checks.get(("tyre_tracks", base)) == "PASS":
                cited_pass.append((ac, base))
            else:
                cited_other.append((ac, base))
print("cited and a PASS line of the re-run (%d):" % len(cited_pass))
by = {}
for ac, n in cited_pass:
    by.setdefault(ac, []).append(n)
for ac in sorted(by):
    print("  %s: %s" % (ac, ", ".join(by[ac])))
print("cited but not a CHECK line of `tyre_tracks` (%d) -- read each by hand:" % len(cited_other))
for ac, n in cited_other:
    print("  %s: %s" % (ac, n))
print("CHECK names in the re-run not cited by any acceptance row:")
cited = {n for _, n in cited_pass}
print("  " + ", ".join(sorted(names - cited)))
