#!/bin/sh
# /story-done 026 on the live tree. Reads the repo, writes only under sd26 (and gdUnit's gitignored reports/).
# The headless half: the tree against the dev session's final manifest, parse, tyre_tracks twice, the showcase headless,
# the 79 older outputs through the exact rewrites, gdUnit. The windowed recordings follow in rec26.sh.
S=/private/tmp/claude-501/-Users-tomas-kunzo-projects-ronin-ya-wasteland-fire
MY=$S/2573aa3b-8b37-4178-bdbd-5cb6686a7ba2/scratchpad/sd26
E26=production/qa/evidence/story-026-tyre-tracks
O=$MY/out
mkdir -p $O
cd /Users/tomas.kunzo/projects/ronin-ya/wasteland-fire || exit 9
date +%H:%M:%S > $O/progress.txt
# 0 the tree against the dev session's final manifest (sorted by path: a locale sort order is not a difference)
sh $MY/manifest.sh $O/manifest-before.txt
LC_ALL=C sort -k2 $O/manifest-before.txt > $O/manifest-before.sorted
LC_ALL=C sort -k2 $E26/measurements/manifest-final.txt > $O/manifest-dev.sorted
diff $O/manifest-before.sorted $O/manifest-dev.sorted > $O/manifest-diff.txt && echo "TREE = THE DEV SESSION'S FINAL MANIFEST ($(wc -l < $O/manifest-dev.sorted) files)" || { echo "TREE DIFFERS FROM THE DEV MANIFEST:"; head -20 $O/manifest-diff.txt; }
# 1 parse every .gd the story's chain parsed
sed 's#^#res://#' $E26/measurements/gd-list.txt | xargs godot --headless --path . -s res://.claude/scripts/godot-parse-check.gd -- > $O/parse.txt 2>&1; echo "parse rc=$?"; grep -c "^ok:" $O/parse.txt; grep "PARSE FAIL" $O/parse.txt
echo "parse done $(date +%H:%M:%S)" >> $O/progress.txt
# 2 the story's scenario twice in parallel, and the showcase headless
for run in a b; do godot --headless --fixed-fps 60 --path . res://tools/evidence/split_screen_harness.tscn -- --scenario=tyre_tracks > $O/tyre_tracks-$run.txt 2>&1 & done; wait
grep -c "^CHECK" $O/tyre_tracks-a.txt; grep -E "^RESULT|FAIL" $O/tyre_tracks-a.txt | cut -c1-300
cmp $O/tyre_tracks-a.txt $O/tyre_tracks-b.txt && echo "tyre_tracks a/b BYTE-IDENTICAL"
cmp $O/tyre_tracks-a.txt $E26/scenario-outputs/split-tyre_tracks.txt && echo "tyre_tracks = the retained output"
godot --headless --fixed-fps 60 --path . res://tools/evidence/split_screen_harness.tscn -- --scenario=tyre_tracks_showcase > $O/showcase-headless.txt 2>&1; echo "showcase rc=$?"; grep -E "^RESULT|FAIL" $O/showcase-headless.txt | cut -c1-200
cmp $O/showcase-headless.txt $E26/scenario-outputs/split-tyre_tracks_showcase.txt && echo "showcase headless = the retained output"
echo "scenarios done $(date +%H:%M:%S)" >> $O/progress.txt
# 3 the 79 older outputs against story 026's baseline, through its exact rewrites
sh $MY/rg/regress.sh live $MY/rg/normalizers26.json 2>&1 | tail -24
echo "regress done $(date +%H:%M:%S)" >> $O/progress.txt
# 4 gdUnit
godot --headless -s -d --remote-debug tcp://127.0.0.1:0 res://addons/gdUnit4/bin/GdUnitCmdTool.gd -a res://tests --ignoreHeadlessMode > $O/gdunit.txt 2>&1; echo "gdunit rc=$?"; grep -a "Overall Summary" $O/gdunit.txt | cut -c1-200
sh $MY/manifest.sh $O/manifest-after.txt
cmp $O/manifest-before.txt $O/manifest-after.txt && echo "TREE UNCHANGED BY THE CHECK"
echo VERIFY_DONE
echo "done $(date +%H:%M:%S) VERIFY_DONE" >> $O/progress.txt
