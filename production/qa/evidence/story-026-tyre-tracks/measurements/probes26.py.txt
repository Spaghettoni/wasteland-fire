"""The strip probe of Story 026: for each strip rule of the runner's _apply_tyre_tracks_off(), copy the project, take that rule out with a
SAME-LINE-COUNT patch of the runner (so the printed advance_ticks frame does not move and the normalizers stay valid), capture the 79
retained outputs there and compare them with the baseline. The outputs that differ are the regression's detectors for that rule.
usage: probes26.py NAME [NAME...]   (names below)"""
import subprocess
import sys
from pathlib import Path

REPO = "/Users/tomas.kunzo/projects/ronin-ya/wasteland-fire"
SP = "/private/tmp/claude-501/-Users-tomas-kunzo-projects-ronin-ya-wasteland-fire/903467d6-8576-4a19-ad15-5c4b7052d4e0/scratchpad"
W = f"{SP}/rg26"
RUNNER = "tools/evidence/split_screen_harness.gd"
OPT = 'if bool(script.get_script_constant_map().get(TYRE_TRACKS_CONSTANT, false)) or root.field == null:\n\t\treturn\n'
PROBES = {
    "tracks_on": [(OPT, 'if true or bool(script.get_script_constant_map().get(TYRE_TRACKS_CONSTANT, false)) or root.field == null:\n\t\treturn\n')],
    "export_kept": [("\troot.field.tyre_tracks = null\n", "\troot.field.tyre_tracks = root.field.tyre_tracks\n")],
}
for name in sys.argv[1:]:
    pairs = PROBES[name]
    root = f"{SP}/probes26/{name}"
    Path(root).mkdir(parents=True, exist_ok=True)
    subprocess.run(["rsync", "-a", "--delete", "--exclude", ".git", "--exclude", "production", "--exclude", "docs", "--exclude", "design",
                    "--exclude", "tests", "--exclude", "assets/art/svaty_jur", REPO + "/", root + "/"], check=True)
    runner = Path(root) / RUNNER
    text = runner.read_text(encoding="utf-8")
    for old, new in pairs:
        assert old.count("\n") == new.count("\n")
        assert text.count(old) == 1, (name, old, text.count(old))
        text = text.replace(old, new)
    runner.write_text(text, encoding="utf-8")
    out = f"{W}/probe-{name}"
    subprocess.run(["sh", f"{W}/capture_in.sh", root, out, f"{W}/jobs.txt", "3"], check=True)
    subprocess.run(["sh", f"{W}/filter.sh", out, f"{W}/f-probe-{name}"], check=True)
    result = subprocess.run(["python3", "-I", f"{W}/compare.py", f"{W}/f-base", f"{W}/f-probe-{name}", f"{W}/normalizers26.json"], capture_output=True, text=True)
    Path(f"{W}/probe-{name}.txt").write_text(result.stdout, encoding="utf-8")
    print(name, "->", [l for l in result.stdout.split("\n") if l.startswith("DIFF") or l.startswith("MISSING") or l.startswith("--")], flush=True)
