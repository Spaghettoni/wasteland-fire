#!/bin/sh
# usage: rec26final.sh   Story 026's retained recordings, one after another (the display awake, nothing else running): the main scene's
# launch (90 frames, Movie Maker, for frame 00 and the byte compare with rec/launch-pre, the tree before the story), then the windowed
# tyre_tracks_showcase in three parts, each with --write-movie into rec/final26-PART/: drive (sand, sand_wide, salt, road, ford, pad, cross, far),
# looks (gyro, wall, restart) and fade (the real minute: TRACKS_FADE_POKE unset).
S=/private/tmp/claude-501/-Users-tomas-kunzo-projects-ronin-ya-wasteland-fire/903467d6-8576-4a19-ad15-5c4b7052d4e0/scratchpad
cd /Users/tomas.kunzo/projects/ronin-ya/wasteland-fire || exit 9
P=$S/rec/final26.progress
date +%H:%M:%S > $P
mkdir -p $S/rec/launch26
godot --path . --windowed --resolution 1280x720 --write-movie $S/rec/launch26/f.png --quit-after 90 > $S/rec/launch26.out 2>&1
echo "launch rc=$? $(date +%H:%M:%S)" >> $P
sh $S/rg26/rec26.sh final26-drive tyre_tracks_showcase TRACKS_SCENES=sand,sand_wide,salt,road,ford,pad,cross,far
echo "drive $(tail -1 $S/rec/final26-drive.progress)" >> $P
sh $S/rg26/rec26.sh final26-looks tyre_tracks_showcase TRACKS_SCENES=gyro,wall,restart
echo "looks $(tail -1 $S/rec/final26-looks.progress)" >> $P
sh $S/rg26/rec26.sh final26-fade tyre_tracks_showcase TRACKS_SCENES=fade
echo "fade $(tail -1 $S/rec/final26-fade.progress)" >> $P
echo ALLDONE >> $P
