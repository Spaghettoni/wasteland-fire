# Mutants no output can tell from the build: reported apart, with the reason, when they survive.
EQUIVALENT = {}
WINDOWED = {}


def prepare(i):
    d = COPIES[i]
    os.makedirs(WORK, exist_ok=True)
    if os.path.exists(d):
        subprocess.run(["find", d, "-delete"], check=True)
    subprocess.run(["rsync", "-a", "--exclude", ".git", "--exclude", "production", "--exclude", "design", "--exclude", "docs",
                    "--exclude", "tests", "--exclude", "assets/art/svaty_jur", REPO + "/", d + "/"], check=True)
    return d


def run(copy, scn, module):
    env = dict(os.environ)
    env["AMMO_ONLY"] = module
    p = subprocess.run(["godot", "--headless", "--fixed-fps", "60", "--path", copy, "res://tools/evidence/split_screen_harness.tscn", "--", "--scenario=" + scn],
                       capture_output=True, text=True, timeout=2400, env=env)
    return p.returncode, p.stdout + p.stderr


def one(copy, mutant):
    name, rel, old, new, scn, kind, module = mutant
    path = os.path.join(copy, rel)
    orig = open(path).read()          # this worker is the only one that touches this copy
    if orig.count(old) != 1:
        return name, None, "NOT APPLIED (text found %d times)" % orig.count(old)
    open(path, "w").write(orig.replace(old, new))
    try:
        rc, out = run(copy, scn, module)
    finally:
        open(path, "w").write(orig)
    if not re.search(r"^RESULT %s " % scn, out, re.M):
        return name, None, "BROKEN: no RESULT line (rc=%d) %s" % (rc, " | ".join(l for l in out.splitlines() if "ERROR" in l)[:300])
    if kind == "check":
        fails = re.findall(r"^CHECK \S+ (\S+) FAIL", out, re.M)
        dead = rc != 0 or bool(fails)
        detail = "%s%s rc=%d fails=%s" % (scn, "[" + module + "]" if module else "", rc, ",".join(fails) or "-")
    else:
        if not os.path.exists(os.path.join(BASE, "split-%s.txt" % scn)):
            return name, None, "BROKEN: no baseline split-%s.txt (the regression kind needs a retained scenario)" % scn
        base = open(os.path.join(BASE, "split-%s.txt" % scn)).read()
        for before, after in RULES.get(scn, []):
            out = out.replace(after, before)
        keep = lambda t: [l for l in t.splitlines() if l.startswith(("SPLIT ", "CHECK ", "RESULT ", "SCRIPT ERROR", "ERROR:", "WARNING:"))]
        dead = keep(out) != keep(base) or rc != 0
        detail = "%s rc=%d output %s the baseline" % (scn, rc, "differs from" if dead else "equals")
    return name, dead, detail


def worker(i, jobs):
    return [one(COPIES[i], m) for m in jobs[i::WORKERS]]


def main():
    if DRY:
        bad = 0
        for name, rel, old, new, scn, kind, module in M:
            n = open(os.path.join(REPO, rel)).read().count(old)
            if n != 1 or old == new:
                bad += 1
                print("DRY  %-52s text found %d times in %s (or no change)" % (name, n, rel))
        print("DRY: %d mutants, %d with a replacement text that is not found exactly once" % (len(M), bad))
        return 1 if bad else 0
    jobs = M
    if "--only" in sys.argv:
        wanted = sys.argv[sys.argv.index("--only") + 1].split(",")
        jobs = [m for m in M if m[0] in wanted]
    for i in range(WORKERS):
        prepare(i)
    with ThreadPoolExecutor(max_workers=WORKERS) as pool:
        results = [r for part in pool.map(lambda i: worker(i, jobs), range(WORKERS)) for r in part]
    order = {m[0]: k for k, m in enumerate(M)}
    results.sort(key=lambda r: order[r[0]])
    killed = survived = broken = 0
    for name, dead, detail in results:
        if dead is False and name in WINDOWED:
            print("WINDOWED   %-52s %s | %s" % (name, detail, WINDOWED[name]))
            killed += 1
            continue
        if dead is False and name in EQUIVALENT:
            print("EQUIVALENT %-52s %s | %s" % (name, detail, EQUIVALENT[name]))
            killed += 1
            continue
        status = "BROKEN " if dead is None else ("KILLED " if dead else "SURVIVED")
        killed += dead is True
        survived += dead is False
        broken += dead is None
        print("%s %-56s %s" % (status, name, detail))
    print("-- %d mutants: %d killed or equivalent, %d survived, %d broken or not applied" % (len(jobs), killed, survived, broken))
    clean = True
    for i in range(WORKERS):
        d = subprocess.run(["diff", "-rq", "--exclude=.godot", "--exclude=production", "--exclude=.git", "--exclude=design", "--exclude=docs",
                            "--exclude=tests", "--exclude=svaty_jur", REPO, COPIES[i]], capture_output=True, text=True)
        if d.stdout.strip():
            clean = False
            print("COPY %d DIFFERS FROM THE REPO:\n%s" % (i, d.stdout[:600]))
    print("-- the %d scratch copies %s the repo (apart from .godot, production, design, docs, tests, svaty_jur)" % (WORKERS, "equal" if clean else "DIFFER FROM"))
    return 0 if survived == 0 and broken == 0 and clean else 1


if __name__ == "__main__":
    sys.exit(main())
