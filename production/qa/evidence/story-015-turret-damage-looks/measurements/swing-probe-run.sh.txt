#!/bin/sh
# usage: swing-probe-run.sh TAG [mutate]   a scratch copy of the repo with the probe registered; windowed run; prints the SPLIT and RESULT lines
R=/Users/tomas.kunzo/projects/ronin-ya/wasteland-fire
D=/private/tmp/claude-501/-Users-tomas-kunzo-projects-ronin-ya-wasteland-fire/011fac19-3980-4a60-9260-2bb6291e87b1/scratchpad/swing
C=$D/copy-$1
mkdir -p $C
rsync -a --delete --exclude .git --exclude production --exclude design --exclude docs $R/ $C/
cp $D/turret_looks_swing_probe.gd $C/tools/evidence/split_screen/
python3 -I - "$C" "$2" <<'PY'
import sys
c, mutate = sys.argv[1], sys.argv[2] if len(sys.argv) > 2 else ""
p = c + "/tools/evidence/split_screen_harness.gd"
s = open(p).read()
old = '	&"turret_looks_fps": preload("res://tools/evidence/split_screen/turret_looks_fps.gd"),\n'
assert s.count(old) == 1
open(p, "w").write(s.replace(old, old + '	&"turret_looks_swing_probe": preload("res://tools/evidence/split_screen/turret_looks_swing_probe.gd"),\n'))
if mutate:
    t = c + "/src/gameplay/defences/turret.gd"
    u = open(t).read()
    old = "		head.reset_physics_interpolation()\n"
    assert u.count(old) == 1
    open(t, "w").write(u.replace(old, ""))
PY
godot --headless --path $C --import > /dev/null 2>&1
cd $C && godot --path . --windowed --resolution 1280x720 res://tools/evidence/split_screen_harness.tscn -- --scenario=turret_looks_swing_probe 2>&1 | grep -E "^SPLIT swing|^RESULT|SCRIPT ERROR|ERROR"
