extends SceneTree
## Probe: instances each Unit type's model scene (from its UnitStats), waits a frame so _ready()
## has built it, and prints its class, children, their groups, materials, triangle counts, the
## model's box against the type's collider, and the hull the rise reads.

const STATS: Array[String] = ["motorbike", "buggy", "truck", "gyrocopter"]


func _initialize() -> void:
	for type_id: String in STATS:
		var stats: UnitStats = load("res://src/gameplay/units/data/%s_stats.tres" % type_id)
		var model: Node3D = stats.model.instantiate()
		root.add_child(model)
		await process_frame
		var script_name: String = (model.get_script() as Script).get_global_name()
		var box: AABB = AABB()
		var first: bool = true
		var tris: int = 0
		var lines: PackedStringArray = []
		for child: Node in model.get_children():
			var mi: MeshInstance3D = child as MeshInstance3D
			if mi == null:
				continue
			var part: AABB = mi.transform * mi.mesh.get_aabb()
			box = part if first else box.merge(part)
			first = false
			var count: int = 0
			for s: int in mi.mesh.get_surface_count():
				var idx: PackedInt32Array = mi.mesh.surface_get_arrays(s)[Mesh.ARRAY_INDEX]
				count += idx.size() / 3
			tris += count
			var mat: Material = mi.material_override
			lines.append("%s team=%s surfaces=%d tris=%d override=%s" % [mi.name,
					mi.is_in_group(Unit.TEAM_COLOUR_GROUP), mi.mesh.get_surface_count(), count,
					mat.resource_name if mat != null and not mat.resource_name.is_empty() else str(mat != null)])
		print("TYPE %s model=%s class=%s scale=%.3f pos=%s tris=%d" % [type_id, stats.model.resource_path.get_file(), script_name, model.scale.x, model.position, tris])
		for line: String in lines:
			print("  ", line)
		var half: Vector3 = stats.collision_size / 2.0
		print("  box x %.2f..%.2f y %.2f..%.2f z %.2f..%.2f | collider x +-%.2f y %.2f..%.2f z +-%.2f" % [
				box.position.x, box.end.x, box.position.y, box.end.y, box.position.z, box.end.z,
				half.x, stats.collision_center.y - half.y, stats.collision_center.y + half.y, half.z])
		var unit_model: UnitModel = model as UnitModel
		if unit_model != null:
			var hull: AABB = unit_model._hull
			print("  hull x %.3f..%.3f y %.3f..%.3f z %.3f..%.3f rise=%s hover=%.3f lowest_over_floor=%.3f" % [
					hull.position.x, hull.end.x, hull.position.y, hull.end.y, hull.position.z, hull.end.z,
					unit_model.rise_over_cliffs, unit_model.hover_height,
					unit_model.hover_height + unit_model.offset.y + hull.position.y * unit_model.model_scale])
		model.free()
	quit()
