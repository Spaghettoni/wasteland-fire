#!/bin/bash
# Rebuilds the three composed frames of Story 025's evidence from the recordings' PNGs and compares them with the retained ones.
#   03e  four consecutive frames of Player 1's rotor (frames 322 to 325 of the recording), each the region 230,370 200x120 at 3x, in a row
#   08   the propeller, standing (frame 562) and at full speed (frame 698), each Player's: the region 262,268 76x50 (Player 1) and
#        902,268 76x50 (Player 2) at 8x, in a row: standing P1, full speed P1, standing P2, full speed P2
#   09   the seeded defect (the smear's trailing side flipped): the same region of frames 186 to 189 of the defect's own recording
S=/private/tmp/claude-501/-Users-tomas-kunzo-projects-ronin-ya-wasteland-fire/d1c9331f-af1b-4190-8906-efd99a380790/scratchpad
cd /Users/tomas.kunzo/projects/ronin-ya/wasteland-fire || exit 9
E=production/qa/evidence/story-025-gyrocopter-rotor
T=$S/docs25/compose-check
mkdir -p $T
C="godot --headless --path . -s $S/lib/compose.gd --"
for n in 322 323 324 325; do $C crop $T/turn_$n.png $S/rec/final/f00000$n.png 230 370 200 120 3 > /dev/null 2>&1; done
$C row $T/03e.png 300 $T/turn_322.png $T/turn_323.png $T/turn_324.png $T/turn_325.png > /dev/null 2>&1
cmp $T/03e.png $E/03e-rotor-four-consecutive-frames-zoomed-3x-player-1.png && echo "03e rebuilt, byte-identical to the retained frame"
for pair in "562 stand" "698 fast"; do
  set -- $pair
  $C crop $T/prop_$2_p1.png $S/rec/final/f00000$1.png 262 268 76 50 8 > /dev/null 2>&1
  $C crop $T/prop_$2_p2.png $S/rec/final/f00000$1.png 902 268 76 50 8 > /dev/null 2>&1
done
$C row $T/08.png 300 $T/prop_stand_p1.png $T/prop_fast_p1.png $T/prop_stand_p2.png $T/prop_fast_p2.png > /dev/null 2>&1
cmp $T/08.png $E/08-propeller-standing-and-at-full-speed-chase-view-zoomed-8x.png && echo "08 rebuilt, byte-identical to the retained frame"
for n in 186 187 188 189; do $C crop $T/flip_$n.png $S/w25f/flipped-rec/f00000$n.png 230 370 200 120 3 > /dev/null 2>&1; done
$C row $T/09.png 300 $T/flip_186.png $T/flip_187.png $T/flip_188.png $T/flip_189.png > /dev/null 2>&1
cmp $T/09.png $E/09-seeded-defect-smear-ahead-of-the-blades-zoomed-3x-player-1.png && echo "09 rebuilt, byte-identical to the retained frame"
