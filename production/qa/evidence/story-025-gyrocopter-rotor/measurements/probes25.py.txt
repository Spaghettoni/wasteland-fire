"""The strip probe of Story 025: for each strip rule of the runner's _apply_rotors_off() (and _without_rotors()), copy the project, take
that rule out with a SAME-LINE-COUNT patch of the runner (so the printed advance_ticks frame does not move and the normalizers stay
valid), capture the 77 retained outputs there and compare them with the baseline. The outputs that differ are the regression's
detectors for that rule. usage: probes25.py NAME [NAME...]   (names below)"""
import subprocess
import sys
from pathlib import Path

REPO = "/Users/tomas.kunzo/projects/ronin-ya/wasteland-fire"
SP = "/private/tmp/claude-501/-Users-tomas-kunzo-projects-ronin-ya-wasteland-fire/d1c9331f-af1b-4190-8906-efd99a380790/scratchpad"
W = f"{SP}/rg25"
RUNNER = "tools/evidence/split_screen_harness.gd"
OPT = 'if bool(script.get_script_constant_map().get(ROTORS_CONSTANT, false)):\n\t\treturn\n'
PROBES = {
    "rotors_on": [(OPT, 'if true or bool(script.get_script_constant_map().get(ROTORS_CONSTANT, false)):\n\t\treturn\n')],
    "file_kept": [("\tmodel.mesh_scene = FROZEN_GYROCOPTER_MESH_FILE\n", "\tmodel.mesh_scene = model.mesh_scene\n")],
    "settings_kept": [("\tmodel.rotor_settings = null\n", "\tmodel.rotor_settings = model.rotor_settings\n")],
    "frozen_scene_left": [("path != SHIPPED_GYROCOPTER_MODEL_PATH and path != FROZEN_GYROCOPTER_MODEL.resource_path:", "path != SHIPPED_GYROCOPTER_MODEL_PATH and true:")],
}
for name in sys.argv[1:]:
    pairs = PROBES[name]
    root = f"{SP}/probes25/{name}"
    Path(root).mkdir(parents=True, exist_ok=True)
    subprocess.run(["rsync", "-a", "--delete", "--exclude", ".git", "--exclude", "production", "--exclude", "docs", "--exclude", "design",
                    "--exclude", "tests", "--exclude", "assets/art/svaty_jur", REPO + "/", root + "/"], check=True)
    runner = Path(root) / RUNNER
    text = runner.read_text(encoding="utf-8")
    for old, new in pairs:
        assert old.count("\n") == new.count("\n")
        assert text.count(old) == 1, (name, old, text.count(old))
        text = text.replace(old, new)
    runner.write_text(text, encoding="utf-8")
    out = f"{W}/probe-{name}"
    subprocess.run(["sh", f"{W}/capture_in.sh", root, out, f"{W}/jobs.txt", "3"], check=True)
    subprocess.run(["sh", f"{W}/filter.sh", out, f"{W}/f-probe-{name}"], check=True)
    result = subprocess.run(["python3", "-I", f"{W}/compare.py", f"{W}/f-base", f"{W}/f-probe-{name}", f"{W}/normalizers25-probe.json"], capture_output=True, text=True)
    Path(f"{W}/probe-{name}.txt").write_text(result.stdout, encoding="utf-8")
    print(name, "->", [l for l in result.stdout.split("\n") if l.startswith("DIFF") or l.startswith("MISSING") or l.startswith("--")], flush=True)
