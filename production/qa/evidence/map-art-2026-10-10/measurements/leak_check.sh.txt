#!/bin/sh
# usage: leak_check.sh   Runs map_showcase headless at 60 fixed fps, cut by --quit-after 1560 as the
# after recording was, in the worktree and in the copy at 73de824, both with --verbose, and prints
# each run's exit messages: whether the leak at exit comes with this work or with the cut.
W=/private/tmp/claude-501/-Users-tomas-kunzo-projects-ronin-ya-wasteland-fire/298d4507-ee5d-4984-a655-a1e248417884/scratchpad/wt
R=/Users/tomas.kunzo/projects/ronin-ya/wasteland-fire/.claude/worktrees/map-art
mkdir -p "$W/leak"
run() {
  cd "$2" || exit 9
  godot --headless --fixed-fps 60 --verbose --path . --quit-after "$3" res://tools/evidence/split_screen_harness.tscn -- --scenario=map_showcase > "$W/leak/$1.txt" 2>&1
  echo "$1 rc $?"
}
run after-1560 "$R" 1560 &
run before-1560 "$W/before-copy" 1560 &
run after-full "$R" 2400 &
wait
for f in after-1560 before-1560 after-full; do
  echo "== $f"
  grep -E "leaked|still in use|^RESULT" "$W/leak/$f.txt"
done
echo "leak check done"
