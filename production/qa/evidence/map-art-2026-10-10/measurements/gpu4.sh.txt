#!/bin/sh
# usage: gpu4.sh   gpu3.sh again with three more pairs, the copy at 73de824 (before) first this time:
# runs map_fps windowed on the Vulkan driver through the scratch tool gpu_time.gd, which measures
# every viewport's GPU and CPU render time per frame, and prints each run's summary and the
# harness's own RESULT line. Nothing else may run meanwhile.
S=/private/tmp/claude-501/-Users-tomas-kunzo-projects-ronin-ya-wasteland-fire/298d4507-ee5d-4984-a655-a1e248417884/scratchpad
W=$S/wt
R=/Users/tomas.kunzo/projects/ronin-ya/wasteland-fire/.claude/worktrees/map-art
mkdir -p "$W/gpu4"
run() { cd "$2" || exit 9; godot --path . --windowed --resolution 1280x720 --disable-vsync --rendering-driver vulkan -s "$S/gpu_time.gd" -- --scenario=map_fps > "$W/gpu4/$1.txt" 2>&1; echo "$1 rc $?"; }
run before-3 "$W/before-copy"
run after-3 "$R"
run before-4 "$W/before-copy"
run after-4 "$R"
run before-5 "$W/before-copy"
run after-5 "$R"
for f in before-3 after-3 before-4 after-4 before-5 after-5; do
  echo "$f: $(grep -h '^GPUTIME_SUMMARY' "$W/gpu4/$f.txt")"
  echo "$f: $(grep -h '^RESULT' "$W/gpu4/$f.txt" | grep -oE 'fps_avg_steady=[0-9.]+|primitives_max=[0-9]+|draw_calls_max=[0-9]+' | tr '\n' ' ')"
done
echo "gpu4 done"
