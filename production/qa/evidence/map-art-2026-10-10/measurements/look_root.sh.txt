#!/bin/sh
# usage: look_root.sh ROOT NAME SCENARIO FRAMES KEEP...   look.sh for any project root: records the
# first FRAMES frames of one evidence scenario windowed at 1280x720 into wt/NAME and keeps the frames
# numbered in KEEP.
W=/private/tmp/claude-501/-Users-tomas-kunzo-projects-ronin-ya-wasteland-fire/298d4507-ee5d-4984-a655-a1e248417884/scratchpad/wt
ROOT="$1"; NAME="$2"; SC="$3"; FRAMES="$4"; shift 4
mkdir -p "$W/$NAME/all"
cd "$ROOT" || exit 9
godot --path . --windowed --resolution 1280x720 --write-movie "$W/$NAME/all/f.png" --quit-after "$FRAMES" \
  res://tools/evidence/split_screen_harness.tscn -- --scenario="$SC" > "$W/$NAME/run.txt" 2>&1
echo "$NAME $SC rc $?"
grep -E "ERROR|SHADER ERROR" "$W/$NAME/run.txt" | head -5
for n in "$@"; do
  src=$(printf "%s/%s/all/f%08d.png" "$W" "$NAME" "$n")
  [ -f "$src" ] && cp "$src" "$W/$NAME/"
done
find "$W/$NAME/all" -name '*.png' -delete
