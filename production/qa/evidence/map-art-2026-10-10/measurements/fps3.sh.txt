#!/bin/sh
# usage: fps3.sh   Runs map_fps windowed with vsync off, alternating the map-art worktree (after) and
# the copy at 73de824 (wt/before-copy, before), twice each, then prints map_fps's own steady average
# and lowest frame rate, draw calls and primitives for each run. Nothing else may run meanwhile.
W=/private/tmp/claude-501/-Users-tomas-kunzo-projects-ronin-ya-wasteland-fire/298d4507-ee5d-4984-a655-a1e248417884/scratchpad/wt
R=/Users/tomas.kunzo/projects/ronin-ya/wasteland-fire/.claude/worktrees/map-art
mkdir -p "$W/fps3"
run() { cd "$2" || exit 9; godot --path . --windowed --resolution 1280x720 --disable-vsync res://tools/evidence/split_screen_harness.tscn -- --scenario=map_fps > "$W/fps3/$1.txt" 2>&1; echo "$1 rc $?"; }
run after-1 "$R"
run before-1 "$W/before-copy"
run after-2 "$R"
run before-2 "$W/before-copy"
for f in after-1 before-1 after-2 before-2; do
  echo "$f: $(grep -h '^RESULT' "$W/fps3/$f.txt" | grep -oE 'fps_avg_steady=[0-9.]+|fps_min_steady=[0-9.]+|draw_calls_max=[0-9]+|primitives_max=[0-9]+' | tr '\n' ' ')"
done
echo "fps3 done"
