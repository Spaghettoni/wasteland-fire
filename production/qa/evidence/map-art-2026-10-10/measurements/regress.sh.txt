#!/bin/sh
# usage: regress.sh TAG   Captures the 54 retained outputs in the map-art worktree (3 at a time) into
# wt/TAG, drops the Jolt job-system load warning (two lines; it appears under parallel load), and
# compares byte for byte with the baseline wt/f-base (the outputs at 73de824, captured in the
# worktree before its first edit). TAG "base" is refused (it would overwrite the baseline); TAG
# "capture-baseline" only captures and filters.
W=/private/tmp/claude-501/-Users-tomas-kunzo-projects-ronin-ya-wasteland-fire/298d4507-ee5d-4984-a655-a1e248417884/scratchpad/wt
TAG="$1"
case "$TAG" in
  ""|base) echo "refusing tag '$TAG': base is the baseline's name"; exit 1 ;;
esac
mkdir -p "$W/$TAG"
xargs -P 3 -L 1 "$W/run_one.sh" "$W/$TAG" < "$W/jobs.txt"
python3 - "$W/$TAG" "$W/f-$TAG" <<'PY'
import sys
from pathlib import Path
src, dst = Path(sys.argv[1]), Path(sys.argv[2])
dst.mkdir(exist_ok=True)
for f in src.iterdir():
    if f.suffix == ".rc":
        (dst / f.name).write_bytes(f.read_bytes())
        continue
    out, skip = [], False
    for line in f.read_bytes().split(b"\n"):
        if skip:
            skip = False
            if line.lstrip().startswith(b"at: CreateJob"):
                continue
        if line.startswith(b"WARNING: Jolt Physics job system exceeded"):
            skip = True
            continue
        out.append(line)
    (dst / f.name).write_bytes(b"\n".join(out))
PY
[ "$TAG" = "capture-baseline" ] && { echo "baseline captured in f-capture-baseline"; exit 0; }
python3 "$W/compare.py" "$W/f-base" "$W/f-$TAG" > "$W/$TAG.txt"
echo "compare exit $?"
grep -v "^STRICT" "$W/$TAG.txt"
