#!/bin/sh
# usage: gpu3.sh   Runs map_fps windowed through the scratch tool gpu_time.gd, which measures every
# viewport's GPU and CPU render time per frame, alternating the map-art worktree (after) and the copy
# at 73de824 (wt/before-copy, before), twice each, and prints each run's summary and the harness's
# own RESULT line. Nothing else may run meanwhile.
S=/private/tmp/claude-501/-Users-tomas-kunzo-projects-ronin-ya-wasteland-fire/298d4507-ee5d-4984-a655-a1e248417884/scratchpad
W=$S/wt
R=/Users/tomas.kunzo/projects/ronin-ya/wasteland-fire/.claude/worktrees/map-art
mkdir -p "$W/gpu3"
run() { cd "$2" || exit 9; godot --path . --windowed --resolution 1280x720 --disable-vsync --rendering-driver vulkan -s "$S/gpu_time.gd" -- --scenario=map_fps > "$W/gpu3/$1.txt" 2>&1; echo "$1 rc $?"; }
run after-1 "$R"
run before-1 "$W/before-copy"
run after-2 "$R"
run before-2 "$W/before-copy"
for f in after-1 before-1 after-2 before-2; do
  echo "$f: $(grep -h '^GPUTIME_SUMMARY' "$W/gpu3/$f.txt")"
  echo "$f: $(grep -h '^RESULT' "$W/gpu3/$f.txt" | grep -oE 'fps_avg_steady=[0-9.]+|primitives_max=[0-9]+|draw_calls_max=[0-9]+' | tr '\n' ' ')"
done
echo "gpu3 done"
