"""usage: glb_triangles.py FILE...   Prints each mesh's triangle count in each glb, from the JSON chunk's
index accessors (or position accessors for meshes without indices)."""
import json
import struct
import sys

for path in sys.argv[1:]:
    data = open(path, "rb").read()
    length = struct.unpack_from("<I", data, 12)[0]
    doc = json.loads(data[20:20 + length])
    total = 0
    rows = []
    for mesh in doc["meshes"]:
        count = 0
        for prim in mesh["primitives"]:
            if "indices" in prim:
                count += doc["accessors"][prim["indices"]]["count"] // 3
            else:
                count += doc["accessors"][prim["attributes"]["POSITION"]]["count"] // 3
        rows.append((mesh.get("name", "?"), count))
        total += count
    print(f"{path}: {len(rows)} meshes, {total} triangles")
    for name, count in rows:
        print(f"  {name:28s} {count}")
