"""Seeded defects for Story 010: each one exact text replacement in a scratch copy of the repo,
run against the scenario that guards it. KILLED when the scenario fails (rc != 0 or a FAIL CHECK),
or, for the regression kind, when an older scenario's output differs from its baseline."""

M = [

def run(scn, kind="check"):
    scene = "res://tools/evidence/drive_harness.tscn" if kind == "drive" else "res://tools/evidence/split_screen_harness.tscn"
    p = subprocess.run(["godot","--headless","--fixed-fps","60","--path",COPY,scene,"--","--scenario="+scn],capture_output=True,text=True,timeout=300)
    return p.returncode, p.stdout + p.stderr
only = sys.argv[1:] 
killed = survived = 0
for name, rel, old, new, scn, kind in M:
    if only and name not in only: continue
    path = os.path.join(COPY, rel)
    orig = open(path).read()
    if orig.count(old) < 1:
        print("MUTANT %-40s NOT APPLIED (text not found)" % name); continue
    if orig.count(old) > 1 and name not in ("round_starts_in_chase",):
        print("MUTANT %-40s NOT APPLIED (text found %d times)" % (name, orig.count(old))); continue
    open(path,"w").write(orig.replace(old, new))
    try:
        rc, out = run(scn, kind)
    finally:
        open(path,"w").write(orig)
    if kind == "check":
        fails = re.findall(r"^CHECK \S+ (\S+) FAIL", out, re.M)
        dead = rc != 0 or bool(fails)
        detail = "%s rc=%d %s fails=%s" % (scn, rc, "FAIL" if dead else "ok", ",".join(fails))
    else:
        base = open(os.path.join(BASE, ("drive-%s.txt" if kind == "drive" else "split-%s.txt") % scn)).read()
        # compare lines that start with the scenario's own output (drop the engine banner)
        keep = lambda text: [l for l in text.splitlines() if l.startswith(("SPLIT ","CHECK ","RESULT ","DRIVE ","WALLS","PINNED","VIOLATION","FPS","SOAK"))]
        dead = keep(out) != keep(base) or rc != 0
        detail = "%s rc=%d output %s the baseline" % (scn, rc, "differs from" if dead else "equals")
    if dead: killed += 1
    else: survived += 1
    print("MUTANT %-40s %s | %s" % (name, "KILLED" if dead else "SURVIVED", detail), flush=True)
print("KILLED=%d SURVIVED=%d" % (killed, survived))
