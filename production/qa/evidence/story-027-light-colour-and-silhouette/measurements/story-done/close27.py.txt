def edit(text, old, new, label):
    n = text.count(old)
    if n != 1:
        raise SystemExit("ANCHOR FAILED (%d matches): %s" % (n, label))
    return text.replace(old, new)

for f in FILES:
    if new[f] == original[f]:
        raise SystemExit("NO CHANGE computed for " + f)
status = subprocess.run(["git", "status", "--porcelain"], cwd=R, capture_output=True, text=True).stdout.strip().split("\n")
print("git status entries:", len(status))
if len(status) != 55:
    raise SystemExit("git status count moved (55 expected): another session may be in the tree")
for f in FILES:
    now = os.stat(os.path.join(R, f)).st_mtime_ns
    if now != mtimes[f]:
        raise SystemExit("MTIME GUARD: %s changed since it was read" % f)
if DRY:
    for f in FILES:
        print("DRY ok  %-70s %d -> %d characters" % (f, len(original[f]), len(new[f])))
    sys.exit(0)
os.makedirs(os.path.join(MY, "backup"), exist_ok=True)
for f in FILES:
    shutil.copyfile(os.path.join(R, f), os.path.join(MY, "backup", os.path.basename(f) + ".before-close"))
for f in FILES:
    with open(os.path.join(R, f), "w", encoding="utf-8", newline="") as fh:
        fh.write(new[f])
    print("written %-70s %d -> %d characters" % (f, len(original[f]), len(new[f])))
