#!/bin/sh
# usage: regress.sh TAG [NORMALIZERS_JSON]   Captures the 81 older outputs (3 at a time) from the repo into sd27/kit/TAG, drops the Jolt job-system load warning from both sides,
# and compares byte for byte with base-a (story 027's baseline, copied from the dev session), through the story's exact rewrites when a file is given.
S=/private/tmp/claude-501/-Users-tomas-kunzo-projects-ronin-ya-wasteland-fire
W=$S/a9ae9e24-5b29-451c-b62f-a16425b4fa8f/scratchpad/sd27/kit
R=/Users/tomas.kunzo/projects/ronin-ya/wasteland-fire
TAG="$1"; NORM="$2"
case "$TAG" in base-a|base-b|base) echo "refusing to overwrite a baseline"; exit 9;; esac
mkdir -p $W/$TAG
sh $W/capture_in.sh $R $W/$TAG $W/jobs.txt 3
sh $W/filter.sh $W/base-a $W/f-base
sh $W/filter.sh $W/$TAG $W/f-$TAG
python3 -I $W/compare.py $W/f-base $W/f-$TAG $NORM > $W/$TAG.txt
echo "compare exit $?"; grep -v "^STRICT" $W/$TAG.txt
