#!/bin/sh
# The final recordings: A (the build as it ships), N (the same moments with the silhouettes taken away), A2 (A again: the frames must equal A's), and the launch of the main scene.
S=/private/tmp/claude-501/-Users-tomas-kunzo-projects-ronin-ya-wasteland-fire/288bd9fe-bc1c-407a-9d36-c535300a0eef/scratchpad
cd /Users/tomas.kunzo/projects/ronin-ya/wasteland-fire || exit 9
date +%H:%M:%S > $S/rec/final.progress
sh $S/rg27/rec27.sh finalA light_showcase X=1
echo "A done $(date +%H:%M:%S)" >> $S/rec/final.progress
sh $S/rg27/rec27.sh finalN light_showcase LIGHT_NO_SIL=1
echo "N done $(date +%H:%M:%S)" >> $S/rec/final.progress
sh $S/rg27/rec27.sh finalA2 light_showcase X=1
echo "A2 done $(date +%H:%M:%S)" >> $S/rec/final.progress
mkdir -p $S/rec/launch
godot --path . --windowed --resolution 1280x720 --write-movie $S/rec/launch/f.png --quit-after 90 > $S/rec/launch.out 2>&1
echo "launch rc=$? $(date +%H:%M:%S)" >> $S/rec/final.progress
echo ALLDONE >> $S/rec/final.progress
