#!/bin/sh
# usage: regress27.sh TAG [NORMALIZERS_JSON]   Captures the 81 older outputs (3 at a time) from the repo into rg27/TAG, drops the Jolt job-system load warning from both sides,
# and compares byte for byte with base-a, through the story's exact rewrites when a file is given.
S=/private/tmp/claude-501/-Users-tomas-kunzo-projects-ronin-ya-wasteland-fire/288bd9fe-bc1c-407a-9d36-c535300a0eef/scratchpad
W=$S/rg27
R=/Users/tomas.kunzo/projects/ronin-ya/wasteland-fire
TAG="$1"; NORM="$2"
case "$TAG" in base-a|base-b|base) echo "refusing to overwrite a baseline"; exit 9;; esac
if [ "$TAG" != "nocapture" ]; then
  mkdir -p $W/$TAG
  sh $W/capture_in.sh $R $W/$TAG $W/jobs.txt 3
fi
sh $W/filter.sh $W/base-a $W/f-base
sh $W/filter.sh $W/$TAG $W/f-$TAG
python3 -I $W/compare.py $W/f-base $W/f-$TAG $NORM > $W/$TAG.txt
echo "compare exit $?"; grep -v "^STRICT" $W/$TAG.txt
