#!/bin/sh
# usage: manifest.sh OUT   A sorted SHA manifest of src, tools, project.godot and assets (the Svaty Jur art left out).
cd /Users/tomas.kunzo/projects/ronin-ya/wasteland-fire || exit 9
find src tools project.godot assets -type f -not -path 'assets/art/svaty_jur/*' -not -name '.DS_Store' | LC_ALL=C sort | xargs shasum > "$1"
wc -l < "$1"
