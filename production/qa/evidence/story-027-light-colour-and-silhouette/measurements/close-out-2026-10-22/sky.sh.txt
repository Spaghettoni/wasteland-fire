#!/bin/bash
# The sky's mean colour above the horizon in the chase-view close-ups under each look (a 160 x 60 crop in each view, under the HUD).
S=/private/tmp/claude-501/-Users-tomas-kunzo-projects-ronin-ya-wasteland-fire/288bd9fe-bc1c-407a-9d36-c535300a0eef/scratchpad
cd /Users/tomas.kunzo/projects/ronin-ya/wasteland-fire || exit 9
E=production/qa/evidence/story-027-light-colour-and-silhouette
for f in 10-close-in-sun-linear-before 14-close-in-sun-agx-without-the-grade 11-close-in-sun-agx-as-shipped 12-close-in-sun-filmic 13-close-in-sun-aces 15-close-in-shade-linear-before 16-close-in-shade-agx-as-shipped; do
  l=$(godot --headless --path . -s $S/lib/measure.gd -- mean $E/$f.png 60 130 160 60 | grep '^MEAN' | cut -d' ' -f3)
  r=$(godot --headless --path . -s $S/lib/measure.gd -- mean $E/$f.png 1100 130 160 60 | grep '^MEAN' | cut -d' ' -f3)

done
