def sh(cmd, cwd=None, timeout=900):
    return subprocess.run(cmd, cwd=cwd, capture_output=True, text=True, timeout=timeout)


def main():
    wanted = set(sys.argv[1:])
    sh(["rsync", "-a", "--delete", "--exclude", ".git/", "--exclude", "builds/", "--exclude", "production/", ROOT + "/", BASE + "/"])
    killed = survived = 0
    for name, path, old, new, scenarios in MUTANTS:
        if wanted and name not in wanted:
            continue
        sh(["rsync", "-a", "--delete", BASE + "/", WORK + "/"])
        text = open(os.path.join(WORK, path)).read()
        if text.count(old) != 1:
            print("MUTANT %-26s NOT APPLIED (%d matches)" % (name, text.count(old)), flush=True)
            continue
        open(os.path.join(WORK, path), "w").write(text.replace(old, new))
        verdicts = []
        dead = False
        for scenario in scenarios:
            r = sh(["godot", "--headless", "--fixed-fps", "60", "--path", WORK, "res://tools/evidence/split_screen_harness.tscn", "--", "--scenario=" + scenario])
            lines = [l for l in r.stdout.splitlines() if l.startswith("RESULT ")]
            fails = [l.split(" ")[2] for l in r.stdout.splitlines() if l.startswith("CHECK ") and " FAIL " in l]
            ok = bool(lines) and " ok " in lines[0] and r.returncode == 0
            dead = dead or not ok
            verdicts.append("%s rc=%s %s fails=%s" % (scenario, r.returncode, "ok" if ok else "FAIL", ",".join(fails) or "-"))
        killed += 1 if dead else 0
        survived += 0 if dead else 1
        print("MUTANT %-26s %s | %s" % (name, "KILLED" if dead else "SURVIVED", " ; ".join(verdicts)), flush=True)
    print("KILLED=%d SURVIVED=%d" % (killed, survived))


main()
