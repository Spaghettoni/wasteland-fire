#!/bin/sh
# usage: frames.sh SCENARIO OUT_DIR   Records one evidence scenario windowed at 1280x720 with Movie Maker
# (read-only use of the repo: frames go under OUT_DIR), keeps only the frames its SPLIT lines name
# (frame=N) in OUT_DIR/kept, deletes the rest, and saves the run log as OUT_DIR/run.txt.
R=/Users/tomas.kunzo/projects/ronin-ya/wasteland-fire
SC="$1"; OUT="$2"
mkdir -p "$OUT/all" "$OUT/kept"
cd "$R" || exit 9
godot --path . --windowed --resolution 1280x720 --write-movie "$OUT/all/f.png" \
  res://tools/evidence/split_screen_harness.tscn -- --scenario="$SC" > "$OUT/run.txt" 2>&1
echo "$SC rc $?"
grep -E "ERROR|WARNING" "$OUT/run.txt" | head -5
grep -E "^RESULT|^CHECK" "$OUT/run.txt" | head -3
grep -o "frame=[0-9]*" "$OUT/run.txt" | sort -u | while read -r f; do
  n=${f#frame=}; src=$(printf "%s/all/f%08d.png" "$OUT" "$n")
  [ -f "$src" ] && cp "$src" "$OUT/kept/"
done
find "$OUT/all" -name '*.png' -delete
echo "$SC kept $(find "$OUT/kept" -name '*.png' | wc -l) frames"
