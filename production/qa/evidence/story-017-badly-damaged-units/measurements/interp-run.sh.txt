#!/bin/sh
# usage: interp-run.sh TAG [mutate]   scratch copy with the probe registered; windowed run; mutate removes the reset in the smoke's _emit()
R=/Users/tomas.kunzo/projects/ronin-ya/wasteland-fire
D=/private/tmp/claude-501/-Users-tomas-kunzo-projects-ronin-ya-wasteland-fire/a697afdf-12e5-4fa4-bf14-518f3d5667d2/scratchpad
C="$D/probe-copy-$1"
mkdir -p "$C"
rsync -a --delete --exclude .git --exclude production --exclude design --exclude docs "$R/" "$C/"
cp "$D/badly_damaged_interp_probe.gd" "$C/tools/evidence/split_screen/"
python3 -I - "$C" "$2" <<'PY'
import sys
c, mutate = sys.argv[1], sys.argv[2] if len(sys.argv) > 2 else ""
p = c + "/tools/evidence/split_screen_harness.gd"
s = open(p).read()
old = '	&"badly_damaged_fps": preload("res://tools/evidence/split_screen/badly_damaged_fps.gd"),\n'
assert s.count(old) == 1
open(p, "w").write(s.replace(old, old + '	&"badly_damaged_interp_probe": preload("res://tools/evidence/split_screen/badly_damaged_interp_probe.gd"),\n'))
if mutate == "toplevel":
    t = c + "/src/gameplay/effects/badly_damaged_smoke.tscn"
    u = open(t).read()
    old = "top_level = true\nvisible = false\n"
    assert u.count(old) == 1
    open(t, "w").write(u.replace(old, "visible = false\n"))
elif mutate:
    t = c + "/src/gameplay/effects/badly_damaged_smoke.gd"
    u = open(t).read()
    old = "	puff.visible = true\n	puff.reset_physics_interpolation()\n"
    assert u.count(old) == 1
    open(t, "w").write(u.replace(old, "	puff.visible = true\n"))
PY
godot --headless --path "$C" --import > /dev/null 2>&1
cd "$C" && godot --path . --windowed --resolution 1280x720 res://tools/evidence/split_screen_harness.tscn -- --scenario=badly_damaged_interp_probe 2>&1 | grep -E "^SPLIT interp|^RESULT|SCRIPT ERROR|ERROR"
