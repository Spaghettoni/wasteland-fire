#!/bin/sh
# The final chain of Story 020: import, parse every .gd the story added or changed, regenerate the exact rewrites from
# the source lines, regress the 66 older outputs, run heights twice (byte compare) and the showcase headless.
S=/private/tmp/claude-501/-Users-tomas-kunzo-projects-ronin-ya-wasteland-fire/f989e592-0c99-4f6a-9d66-b271794c14c4/scratchpad
cd /Users/tomas.kunzo/projects/ronin-ya/wasteland-fire || exit 9
godot --headless --path . --import > $S/w20/import.txt 2>&1; echo "import rc=$?"; grep -c -E "ERROR|WARNING" $S/w20/import.txt
{ for f in src/gameplay/units/unit.gd src/gameplay/units/models/unit_model.gd src/gameplay/units/models/mesh_unit_model.gd src/gameplay/units/models/kit_unit_model.gd tools/evidence/split_screen_harness.gd; do echo $f; done; git ls-files --others --exclude-standard -- 'tools/evidence/split_screen/heights*.gd' | grep '\.gd$'; } | sort -u > $S/w20/gd-list.txt
wc -l < $S/w20/gd-list.txt
sed 's#^#res://#' $S/w20/gd-list.txt | xargs godot --headless --path . -s res://.claude/scripts/godot-parse-check.gd -- > $S/w20/parse.txt 2>&1; echo "parse rc=$?"; grep -c "^ok:" $S/w20/parse.txt; grep "PARSE FAIL" $S/w20/parse.txt
python3 -I $S/rg20/mk_normalizers.py /Users/tomas.kunzo/projects/ronin-ya/wasteland-fire $S/rg20/f-base $S/rg20/before20 $S/rg20/normalizers20.json | tail -3
sh $S/rg20/regress.sh final $S/rg20/normalizers20.json 2>&1 | tail -14
( for run in a b; do godot --headless --fixed-fps 60 --path . res://tools/evidence/split_screen_harness.tscn -- --scenario=heights > $S/gsw20/final-$run.txt 2>&1 & done; wait )
grep -c "^CHECK" $S/gsw20/final-a.txt; grep -E "^RESULT|FAIL" $S/gsw20/final-a.txt | cut -c1-300
cmp $S/gsw20/final-a.txt $S/gsw20/final-b.txt && echo "BYTE-IDENTICAL"
godot --headless --fixed-fps 60 --path . res://tools/evidence/split_screen_harness.tscn -- --scenario=heights_showcase > $S/gsw20/showcase-headless.txt 2>&1; grep -E "^RESULT|FAIL" $S/gsw20/showcase-headless.txt | cut -c1-200
echo FINAL_DONE
