extends SceneTree
## Scratch probe for Story 020 (never in the repo): every collider of Map 01 with its layer, its top (shape box) and its
## drawn top, the drawn tops of every Unit type (thin parts apart), the Gyrocopter's drawn extent in its file, and every
## drawn prop of the dressing that stands above 2.0 m.

const THIN: Dictionary = {
	"buggy": [["whip and pennant", AABB(Vector3(0.38, 1.2, 0.95), Vector3(0.08, 0.85, 0.6))]],
	"truck": [["whip antenna", AABB(Vector3(-0.82, 1.5, 0.2), Vector3(0.14, 1.25, 0.25))]],
}
const LOW_MASK: int = 32 | 64
const HIGH_MASK: int = 1


func _initialize() -> void:
	_units()
	var map: Node3D = (load("res://src/gameplay/maps/map_01/map_01.tscn") as PackedScene).instantiate() as Node3D
	root.add_child(map)
	await process_frame
	await process_frame
	print("--- COLLIDERS of Map 01 (layer, shape top, drawn top) ---")
	var rows: Array = []
	_walk(map, rows)
	for row: Array in rows:
		print("COLLIDER layer=%-3d kind=%-6s shape_top=%7.3f drawn_top=%7.3f path=%s" % row)
	print("--- DRAWN ABOVE 2.0 m with no collider (dressing and others) ---")
	var count: int = 0
	_dressing(map, count)
	quit()


func _units() -> void:
	var gyro_low: float = 0.0
	for type_id: String in ["motorbike", "buggy", "truck", "gyrocopter"]:
		var stats: UnitStats = load("res://src/gameplay/units/data/%s_stats.tres" % type_id)
		var model: MeshUnitModel = stats.model.instantiate() as MeshUnitModel
		var file: PackedScene = model.mesh_scene
		var all: AABB = MeshUnitModel.hull_of(file, INF)
		var meshes: Dictionary = MeshFile.meshes_of(file)
		var body_top: float = -INF
		var thin_top: Dictionary = {}
		var boxes: Array = THIN.get(type_id, [])
		for mesh_name: StringName in meshes:
			var mesh: Mesh = meshes[mesh_name]
			for surface: int in mesh.get_surface_count():
				for point: Vector3 in mesh.surface_get_arrays(surface)[Mesh.ARRAY_VERTEX]:
					var in_thin: String = ""
					for thin: Array in boxes:
						if (thin[1] as AABB).has_point(point):
							in_thin = thin[0]
					if in_thin.is_empty():
						body_top = maxf(body_top, point.y)
					else:
						thin_top[in_thin] = maxf(thin_top.get(in_thin, -INF), point.y)
		print("UNIT %-10s file y %.4f..%.4f  body_top %.4f  thin %s  hover=%.3f hull_top=%s scale=%.3f offset=%s" % [
				type_id, all.position.y, all.end.y, body_top, thin_top, model.hover_height, model.hull_top, model.model_scale, model.offset])
		if type_id == "gyrocopter":
			var body: AABB = MeshUnitModel.hull_of(file, model.hull_top)
			print("  gyrocopter body hull (y <= hull_top %.2f): y %.4f..%.4f x %.3f..%.3f z %.3f..%.3f ; rotor top %.4f" % [
					model.hull_top, body.position.y, body.end.y, body.position.x, body.end.x, body.position.z, body.end.z, all.end.y])
		model.free()
	var kit_stats: PackedScene = load("res://src/gameplay/units/models/gyrocopter_kit_model.tscn")
	print("kit scene loads: %s" % [kit_stats != null])


func _kind(path: String) -> String:
	if "/WallPanel" in path or "Panel" in path.get_file() or "/Walls/" in path or "/Garage/" in path or "FieldWalls" in path:
		return "panel"
	if "FlagWall" in path:
		return "flagw"
	if "Cliff" in path or "cliff" in path:
		return "cliff"
	if "Water" in path or "water" in path or "Channel" in path:
		return "water"
	if "Cover" in path:
		return "cover"
	return "other"


func _drawn_top(node: Node) -> float:
	if node is GeometryInstance3D:
		return ((node as GeometryInstance3D).global_transform * (node as GeometryInstance3D).get_aabb()).end.y
	var top: float = -INF
	for child: Node in node.get_children():
		if child is GeometryInstance3D:
			top = maxf(top, ((child as GeometryInstance3D).global_transform * (child as GeometryInstance3D).get_aabb()).end.y)
	return top


func _walk(node: Node, rows: Array) -> void:
	if node is CollisionObject3D:
		var body: CollisionObject3D = node as CollisionObject3D
		var shape_top: float = -INF
		for child: Node in body.get_children():
			var cs: CollisionShape3D = child as CollisionShape3D
			if cs != null and cs.shape != null and not cs.disabled:
				var local: AABB = cs.shape.get_debug_mesh().get_aabb()
				shape_top = maxf(shape_top, (cs.global_transform * local).end.y)
		rows.append([body.collision_layer, _kind(str(body.get_path())), shape_top, _drawn_top(body), str(body.get_path())])
	elif node is CSGShape3D and (node as CSGShape3D).use_collision and (node as CSGShape3D).is_root_shape():
		var csg: CSGShape3D = node as CSGShape3D
		var box: AABB = csg.global_transform * csg.get_aabb()
		rows.append([csg.collision_layer, "csg", box.end.y, box.end.y, str(csg.get_path())])
	for child: Node in node.get_children():
		_walk(child, rows)


func _dressing(node: Node, count: int) -> void:
	if node is GeometryInstance3D and (node.get_parent() is not CollisionObject3D):
		var top: float = ((node as GeometryInstance3D).global_transform * (node as GeometryInstance3D).get_aabb()).end.y
		if top > 2.0 and (node as GeometryInstance3D).layers & 1 != 0 and not str(node.get_path()).contains("WaterTower") and not str(node.get_path()).contains("Walls"):
			print("DRAWN top=%7.3f layers=%d path=%s" % [top, (node as GeometryInstance3D).layers, node.get_path()])
	for child: Node in node.get_children():
		_dressing(child, count)
